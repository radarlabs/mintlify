description: "Assess device and network risk before a sensitive action, and ask for a verified location check on your backend only when the risk is elevated."
---

By the end of this guide, a sensitive action in your app, such as a payment or a cashout, goes through a risk assessment first. When Radar reports a risk level of `none` or `low`, the action continues. When the level is anything else, or no assessment is available, the user must pass a verified location check before your backend authorizes the action.

If the action always requires the user to be in a particular place, regardless of risk, you don't need the risk step. See [Verify location for non-gaming use cases](/geo-compliance/non-gaming), which runs the same check as a fixed condition.

## Complete example

This is the whole client flow for one withdrawal, and the sections below explain each part. `api` is your own backend client, which calls back on the main thread, and the `show` functions are your UI.

<CodeGroup>

```swift Swift
import CoreLocation
import RadarSDK

// At app launch
Radar.initialize(publishableKey: "prj_test_pk_...")

final class WithdrawalFlow: NSObject, CLLocationManagerDelegate {
  private var locationManager: CLLocationManager?
  private var transactionId = ""

  func withdraw(userId: String, amount: Int) {
    Radar.setUserId(userId)
    api.createWithdrawal(amount: amount) { transactionId in
      self.transactionId = transactionId

      Radar.revealRisk { status, risk in
        // A nil token tells your backend that the risk is unknown.
        let riskToken = status == .success ? risk?.token : nil
        api.sendRiskToken(riskToken, transactionId: transactionId) { result in
          if result.authorized { showDone() } else { self.requestLocation() }
        }
      }
    }
  }

  // Elevated or unknown risk: your backend asks for a verified location check.
  private func requestLocation() {
    // A new manager reports the current status to locationManagerDidChangeAuthorization.
    locationManager = CLLocationManager()
    locationManager?.delegate = self
  }

  func locationManagerDidChangeAuthorization(_ manager: CLLocationManager) {
    switch manager.authorizationStatus {
    case .notDetermined: manager.requestWhenInUseAuthorization()
    case .authorizedWhenInUse, .authorizedAlways: verifyLocation()
    default: showLocationNeeded()
    }
  }

  private func verifyLocation() {
    locationManager = nil
    Radar.trackVerified(
      beacons: false,
      desiredAccuracy: .medium,
      reason: "withdrawal",
      transactionId: transactionId
    ) { status, token in
      guard status == .success, let jwt = token?.token else {
        // The check didn't complete. No token was issued.
        showTryAgain(status)
        return
      }
      api.sendGeolocationToken(jwt, transactionId: self.transactionId) { result in
        if result.authorized { showDone() } else { showFailed(result.failureReasons) }
      }
    }
  }
}
```

```kotlin Kotlin
import android.Manifest
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import io.radar.sdk.Radar
import io.radar.sdk.RadarInitializeOptions
import io.radar.sdk.RadarTrackingOptions

// In Application.onCreate()
Radar.initialize(
  context = this,
  publishableKey = "prj_test_pk_...",
  options = RadarInitializeOptions(fraud = true)
)

class WithdrawalActivity : AppCompatActivity() {
  private var transactionId = ""

  private val locationPermission = registerForActivityResult(
    ActivityResultContracts.RequestMultiplePermissions()
  ) { grants ->
    if (grants.values.any { it }) verifyLocation() else showLocationNeeded()
  }

  fun withdraw(userId: String, amount: Int) {
    Radar.setUserId(userId)
    api.createWithdrawal(amount) { id ->
      transactionId = id

      Radar.revealRisk { status, risk ->
        // A null token tells your backend that the risk is unknown.
        val riskToken = if (status == Radar.RadarStatus.SUCCESS) risk?.token else null
        api.sendRiskToken(id, riskToken) { result ->
          if (result.authorized) showDone() else requestLocation()
        }
      }
    }
  }

  // Elevated or unknown risk: your backend asks for a verified location check.
  private fun requestLocation() {
    locationPermission.launch(
      arrayOf(
        Manifest.permission.ACCESS_FINE_LOCATION,
        Manifest.permission.ACCESS_COARSE_LOCATION
      )
    )
  }

  private fun verifyLocation() {
    Radar.trackVerified(
      beacons = false,
      desiredAccuracy = RadarTrackingOptions.RadarTrackingOptionsDesiredAccuracy.MEDIUM,
      reason = "withdrawal",
      transactionId = transactionId
    ) { status, token ->
      if (status != Radar.RadarStatus.SUCCESS || token == null) {
        // The check didn't complete. No token was issued.
        showTryAgain(status)
        return@trackVerified
      }
      api.sendGeolocationToken(transactionId, token.token) { result ->
        if (result.authorized) showDone() else showFailed(result.failureReasons)
      }
    }
  }
}
```

```javascript JavaScript
import Radar from 'radar-sdk-js';
import { createFraudPlugin } from '@radarlabs/plugin-fraud';

// When the page loads
Radar.registerPlugin(createFraudPlugin());
Radar.initialize('prj_test_pk_...');

export async function withdraw(userId, amount) {
  Radar.setUserId(userId);
  const { transactionId } = await api.createWithdrawal(amount);

  let riskToken = null;
  try {
    riskToken = (await Radar.fraud.revealRisk()).token ?? null;
  } catch (err) {
    // The assessment didn't complete. Your backend treats the risk as unknown.
  }

  let result = await api.sendRiskToken(transactionId, riskToken);
  if (result.authorized) return showDone();

  // Elevated or unknown risk: your backend asks for a verified location check.
  // The browser shows its location prompt when the check runs.
  let geolocationToken;
  try {
    const check = await Radar.fraud.trackVerified({
      skipVerifyApp: true,
      reason: 'withdrawal',
      transactionId,
    });
    geolocationToken = check.token;
  } catch (err) {
    // The check didn't complete. No token was issued.
    return err.status === 'ERROR_PERMISSIONS' ? showLocationNeeded() : showTryAgain(err.status);
  }

  result = await api.sendGeolocationToken(transactionId, geolocationToken);
  return result.authorized ? showDone() : showFailed(result.failureReasons);
}
```

```javascript React Native
import Radar from 'react-native-radar';

// At app launch. The second argument turns on fraud detection.
Radar.initialize('prj_test_pk_...', true);

  if (permission !== 'GRANTED_FOREGROUND' && permission !== 'GRANTED_BACKGROUND') {

    // The check didn't complete. No token was issued.
    return showTryAgain(err);
  }
  if (!geolocationToken) return showTryAgain(null);

  result = await api.sendGeolocationToken(transactionId, geolocationToken);
```

Your backend creates the transaction, verifies both tokens with your project's JWT secret key, and authorizes the withdrawal. `requireSession`, `transactions`, and `releaseWithdrawal` are your own session middleware, store, and downstream call.

<CodeGroup>

```javascript Node.js
import { randomUUID } from 'node:crypto';
import express from 'express';
import { jwtVerify } from 'jose';

const app = express();
app.use(express.json());

const secret = new TextEncoder().encode(process.env.RADAR_JWT_SECRET_KEY);

// Checks the signature and the expiry, and accepts HS256 only.
async function verifyRadarToken(token) {
  const { payload } = await jwtVerify(token ?? '', secret, { algorithms: ['HS256'] });
  return payload;
}

app.post('/api/withdrawals', requireSession, async (req, res) => {
  const transaction = await transactions.create({
    id: `txn-${randomUUID()}`,
    userId: req.session.userId,
    amount: req.body.amount,
    status: 'awaiting_risk',
    expiresAt: Date.now() + 10 * 60 * 1000,
  });
  res.status(201).json({ transactionId: transaction.id });
});

app.post('/api/withdrawals/:id/risk', requireSession, async (req, res) => {
  const transaction = await transactions.getOpen(req.params.id, req.session.userId, 'awaiting_risk');
  if (!transaction) return res.status(409).json({ authorized: false });

  let risk = null;
  try {
    const assessment = await verifyRadarToken(req.body.riskToken);
    // The token has no user ID. Record its jti against this transaction, once.
    if (await transactions.recordRiskAssessment(transaction.id, assessment.jti)) {
      risk = assessment.risk;
    }
  } catch (err) {
    // Missing, changed, or expired. The risk stays unknown.
  }

  if (risk?.level === 'none' || risk?.level === 'low') {
    if (await transactions.transition(transaction.id, 'awaiting_risk', 'authorized')) {
      await releaseWithdrawal(transaction.id);
    }
    return res.json({ authorized: true });
  }

  // Medium, high, or unknown risk: ask for a verified location check.
  await transactions.transition(transaction.id, 'awaiting_risk', 'awaiting_location');
  res.json({ authorized: false, locationRequired: true });
});

app.post('/api/withdrawals/:id/location', requireSession, async (req, res) => {
  const transaction = await transactions.getOpen(req.params.id, req.session.userId, 'awaiting_location');
  if (!transaction) return res.status(409).json({ authorized: false });

  let check;
  try {
    check = await verifyRadarToken(req.body.geolocationToken);
  } catch (err) {
    return res.status(400).json({ authorized: false, error: 'token_invalid' });
  }

  // The check must be for this user, this transaction, and this environment.
  const live = process.env.NODE_ENV !== 'production' || check.user?.live === true;
  if (check.user?.userId !== req.session.userId || check.transactionId !== transaction.id || !live) {
    return res.status(400).json({ authorized: false, error: 'token_mismatch' });
  }

  if (check.passed !== true) {
    return res.status(403).json({
      authorized: false,
      passed: false,
      failureReasons: check.failureReasons ?? [],
    });
  }

  if (await transactions.transition(transaction.id, 'awaiting_location', 'authorized')) {
    await releaseWithdrawal(transaction.id);
  }
  res.json({ authorized: true });
});
```

## Prerequisites

- The device and network risk assessment must be enabled for your project. Contact your Radar account team.
- The Radar SDK and fraud plugin for your platform. See [Fraud plugin setup](/sdk/fraud). `Radar.revealRisk()` requires iOS SDK 3.38.0 or later, Android SDK 3.35.0 or later, `@radarlabs/plugin-fraud` 1.4.0 or later, or React Native SDK 4.36.0 or later.
- A Test publishable key in your app, and the project's JWT secret key on your backend only. See [Choose the right API key](/fundamentals/authentication-and-access#choose-the-right-api-key).
- An authenticated user, with `Radar.setUserId()` called with that user's ID before the verified location check. See [Handle login, logout, and user switching](/fundamentals/authentication-and-access#handle-login-logout-and-user-switching).

## Create the transaction on your backend

Create the transaction before any Radar call, on an authenticated request, as the `POST /api/withdrawals` route does. Generate the transaction ID on your backend so your app can't choose it, and give the transaction a time limit that fits the action. The example uses 10 minutes.

The transaction waits for a risk result (`awaiting_risk`), then either becomes `authorized` or waits for a location check (`awaiting_location`). The `transactions` store is yours. Its `transition` operation must be a compare-and-set, so two concurrent requests can't both authorize.

## Assess risk and branch on the risk level

In your app, call `Radar.revealRisk()` and send the signed token with the transaction ID. If the call fails or returns no token, still call your backend without one.

On your backend, verify the token before you read it, then read `risk.level`. The `/risk` route has the only branch in this guide: `none` or `low` authorizes the action, and anything else requires the verified location check. Unknown risk, from a missing, invalid, expired, or reused token, takes the location path, because unknown risk isn't low risk.

The risk token is signed with `HS256` and your project's JWT secret key, and it carries no user ID. It is tied to the user and the transaction only because your backend received it on that user's authenticated request and recorded its `jti` once.

A risk level isn't a probability of fraud, and there is no universal threshold. Which levels need the location check is your choice for each action. See [Understand risk scores and decisions](/fraud/overview#understand-risk-scores-and-decisions).

## Ask for location permission

When your backend asks for the location check, tell the user why you need their location for this action, then request foreground location permission, as each tab of the complete example does. On the web, the browser shows its own prompt when the check runs. See [Understand foreground and background location permissions](/fundamentals/location-permissions).

Granting permission isn't verification. The transaction stays in `awaiting_location` until a check passes and your backend verifies it.

## Request the verified location check

Call `Radar.trackVerified()` with the transaction ID from your backend as `transactionId`, and a `reason` from a small, fixed set, such as `withdrawal`. Radar returns both in the token and shows the reason in the Radar dashboard. Keep the same `transactionId` for every retry of the same transaction.

On the web, pass `skipVerifyApp: true`, which runs the check in the browser without the Radar Verify companion app.

The check returns a signed geolocation token whether it passed or failed. Send the token to your backend either way. When the check returns an error status, there is no token, and nothing was verified.

To compare the location with an address you hold for the user, call `Radar.setExpectedAddress()` before the check. Address verification must be enabled for your project. For the result, see [Expected address fields](/api/geolocation-tokens#expected-address-fields).

## Validate the token and authorize once

The geolocation token is signed with the same key as the risk token. The `/location` route verifies its signature, algorithm, and expiry, and then checks, in order:

1. `user.userId` is the session user, and `transactionId` is this transaction.
2. In production, `user.live` is `true`.
3. `passed` is `true`. Otherwise it responds with `failureReasons` and leaves the transaction in `awaiting_location`.

Only then does it move the transaction to `authorized`, with a compare-and-set, and call your downstream system. Never authorize from a `passed` value that your app sends outside a verified token. For the full procedure, see [Validate a token](/api/geolocation-tokens#validate-a-token).

## Handle results and errors

No row authorizes the action on the client. In every row except the last, the transaction stays in `awaiting_location`, and a retry uses the same `transactionId`.

| Result or error | What it means | What to do |
| --- | --- | --- |
| No valid risk token | The assessment failed, or the token is invalid, expired, or already used. | Require the verified location check. |
| Permission denied | The user declined the prompt, or the check returned `ERROR_PERMISSIONS`. | Explain why location is needed and how to turn it on in settings, and offer an alternative such as support. |
| Location unavailable | The check returned `ERROR_LOCATION`. | Ask the user to turn on location services and Wi-Fi, and let them try again. |
| Technical error | The check returned `ERROR_NETWORK`, `ERROR_SERVER`, or another error status, and no token. It isn't a failed check and not a pass. | Ask the user to check the connection and try again. |
| Cancelled | The user left before a check completed. Radar sends no signal for this. | Let the transaction expire at its time limit. The user starts a new one. |
| Token expired | Your backend's verification fails because `exp` has passed. | Respond with `400`. Your app runs a new check and resends. |
| Token for another user or transaction | `user.userId` or `transactionId` doesn't match. | Respond with `400`. Your app sets the correct user ID and runs a new check. |
| Failed check | A verified token with `passed` set to `false`. | Respond with `403` and `failureReasons`. Let the user fix the cause, such as disconnecting a VPN, before a new check. Don't loop. |

## Test the integration

Run these in the Test environment with a Test publishable key.

- **Passing case**: Run the flow on a device with location permission granted. Confirm that your backend responds with `authorized` set to `true`, from the risk route when the level is `none` or `low` and from the location route otherwise. Send the same passing token again, and confirm a `409`.
- **Failing case**: Call the risk route with no token, and confirm that `locationRequired` is `true`. Then block your test user with a block rule and run the check. Confirm a `403` with `passed` set to `false` and `fraud_blocked_user_id` in `failureReasons`.

To create and remove the block rule, see [Force a pass or a fail](/fundamentals/testing-and-launch#force-a-pass-or-a-fail). Bypass rules are for internal testing only.

## Production checklist

- [ ] Your backend creates the transaction ID. Your app never chooses it.
- [ ] Both tokens are verified on your backend with the JWT secret key and an explicit `HS256` allowlist.
- [ ] A missing, invalid, or expired risk token leads to the verified location check, not to direct authorization.
- [ ] The geolocation token is bound to the authenticated user and to the transaction ID.
- [ ] The transaction is authorized at most once, with an atomic state change, and never from a client-side value.
- [ ] Production builds use a Live publishable key, and your backend rejects tokens with `user.live` set to `false`.
