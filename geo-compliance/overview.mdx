Geo-compliance verifies that a user is in a location where your product is allowed, and that the location can be trusted. At the end of this quickstart, your app runs one verified location check, your backend verifies the returned geolocation token and reads `passed`, and you have seen both a passed and a failed check.

<Info>
  Fraud is available on the [Enterprise plan](https://radar.com/pricing).
</Info>

## Quickstart

Before you start:

- Enable Fraud for your project: under **Fraud** on the [Settings](https://dashboard.radar.com/settings#fraud-settings) page of the Radar dashboard, select **Yes** for **Enabled**. In the same section, allow the state you are testing from in the country and state lists. An empty list allows every country or state.
- Install the Radar SDK and the fraud plugin for your platform. See [Fraud plugin setup](/sdk/fraud). The Mac and Windows SDKs are available on request.
- Get a Test publishable key for your app, and your project's JWT secret key for your backend. The JWT secret key is shown in the same **Fraud** section. See [Choose the right API key](/fundamentals/authentication-and-access#choose-the-right-api-key).

The client initializes the SDK, sets the signed-in user, requests location permission, runs one check, and sends the token to your backend:

<CodeGroup>

```swift Swift
import CoreLocation
import RadarSDK

final class LocationCheck: NSObject, CLLocationManagerDelegate {
  private let locationManager = CLLocationManager()

  func start() {
    Radar.initialize(publishableKey: "prj_test_pk_...")
    Radar.setUserId("user-123")
    Radar.setProduct("sportsbook")

    locationManager.delegate = self
    locationManager.requestWhenInUseAuthorization()
  }

  func locationManagerDidChangeAuthorization(_ manager: CLLocationManager) {
    let status = manager.authorizationStatus
    if status == .authorizedWhenInUse || status == .authorizedAlways {
      check(reason: "login")
    }
  }

  func check(reason: String) {
    Radar.trackVerified(
      beacons: false,
      desiredAccuracy: .medium,
      reason: reason,
      transactionId: nil
    ) { (status, token) in
      guard status == .success, let token = token, let jwt = token.token else {
        showCheckError(status)
        return
      }

      if token.passed {
        sendTokenToBackend(jwt)
      } else {
        showFailure(token.failureReasons ?? [])
      }
    }
  }
}
```

```kotlin Kotlin
import android.Manifest
import android.os.Bundle
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import io.radar.sdk.Radar
import io.radar.sdk.RadarInitializeOptions

class MainActivity : AppCompatActivity() {
  private val requestLocation =
    registerForActivityResult(ActivityResultContracts.RequestMultiplePermissions()) { grants ->
      if (grants.values.any { it }) check("login")
    }

  override fun onCreate(savedInstanceState: Bundle?) {
    super.onCreate(savedInstanceState)

    Radar.initialize(
      context = this,
      publishableKey = "prj_test_pk_...",
      options = RadarInitializeOptions(fraud = true)
    )
    Radar.setUserId("user-123")
    Radar.setProduct("sportsbook")

    requestLocation.launch(
      arrayOf(
        Manifest.permission.ACCESS_FINE_LOCATION,
        Manifest.permission.ACCESS_COARSE_LOCATION
      )
    )
  }

  private fun check(reason: String) {
    Radar.trackVerified(reason = reason) { status, token ->
      if (status != Radar.RadarStatus.SUCCESS || token == null) {
        showCheckError(status)
        return@trackVerified
      }

      if (token.passed) {
        sendTokenToBackend(token.token)
      } else {
        showFailure(token.failureReasons)
      }
    }
  }
}
```

```javascript JavaScript
      showFailure(result.failureReasons);

    if (err instanceof Radar.fraud.errors.RadarVerifyAppError) {

      showCheckError(err.status);

  }
```

The backend verifies the token and reads `passed`:

<CodeGroup>

```javascript Node.js
import express from 'express';
import { jwtVerify } from 'jose';
import { requireSession } from './auth.js';

const secret = new TextEncoder().encode(process.env.RADAR_JWT_SECRET_KEY);
const app = express();
app.use(express.json());

app.post('/api/wagers', requireSession, async (req, res) => {
  let payload;
  try {
    ({ payload } = await jwtVerify(req.body.geolocationToken, secret, {
      algorithms: ['HS256'],
    }));
  } catch (err) {
    return res.status(401).json({ error: 'invalid_geolocation_token' });
  }

  if (payload.user?.userId !== req.session.userId || payload.product !== 'sportsbook') {
    return res.status(401).json({ error: 'invalid_geolocation_token' });
  }

  if (payload.passed !== true) {
    return res.status(403).json({ passed: false, failureReasons: payload.failureReasons });
  }

  // Place the wager.
  res.status(201).json({ placed: true });
});
```

```python Python
import os

import jwt  # PyJWT
from flask import Flask, jsonify, request, session

app = Flask(__name__)
app.secret_key = os.environ["SESSION_SECRET_KEY"]
SECRET = os.environ["RADAR_JWT_SECRET_KEY"]


@app.post("/api/wagers")
def place_wager():
    if "user_id" not in session:
        return jsonify(error="not_signed_in"), 401

    body = request.get_json(silent=True) or {}
    try:
        payload = jwt.decode(body.get("geolocationToken", ""), SECRET, algorithms=["HS256"])
    except jwt.InvalidTokenError:
        return jsonify(error="invalid_geolocation_token"), 401

    user_id = (payload.get("user") or {}).get("userId")
    if user_id != session["user_id"] or payload.get("product") != "sportsbook":
        return jsonify(error="invalid_geolocation_token"), 401

    if payload.get("passed") is not True:
        return jsonify(passed=False, failureReasons=payload.get("failureReasons", [])), 403

    # Place the wager.
    return jsonify(placed=True), 201
```

</CodeGroup>

Your backend decides, not your app. The `passed` value on the client is for your UI. For every validation step, see [Validate a token](/api/geolocation-tokens#validate-a-token).

## See a passed and a failed result

Sign in as `user-123` from an allowed state and trigger the action. After signature verification, the payload includes these fields, and your backend performs the action:

```json
{
  "passed": true,
  "failureReasons": [],
  "product": "sportsbook",
  "reason": "login",
  "user": {
    "userId": "user-123",
    "state": { "code": "NJ", "allowed": true, "passed": true },
    "fraud": { "passed": true, "verified": true }
  }
}
```

To see a failed result without changing location, create a rule on the **Block rules** page in the dashboard that blocks the user ID `user-123`, then trigger the action again. The request succeeds, the token carries a failed result, and your backend rejects the action:

```json
{
  "passed": false,
  "failureReasons": ["fraud_blocked_user_id"],
  "product": "sportsbook",
  "reason": "login",
  "user": {
    "userId": "user-123",
    "state": { "code": "NJ", "allowed": true, "passed": true },
    "fraud": { "passed": false, "verified": true, "blocked": true }
  }
}
```

Remove the block rule when you finish. See [Force a pass or a fail](/fundamentals/testing-and-launch#force-a-pass-or-a-fail), and for the full payload, [Geolocation token schema](/api/geolocation-tokens).

## Understand check results

A check produces four separate facts. They aren't interchangeable:

- **The request succeeded**: The client returned a success status and a token. The token can still carry `passed` set to `false`.
- **The token is authentic**: Your backend verified the signature and expiry with your JWT secret key. That doesn't say the token belongs to this user, product, or action.
- **The location was verified**: `passed` is `true` in the verified payload, so the jurisdiction and fraud checks passed.
- **The user is eligible**: The token is authentic, belongs to the signed-in user and the intended product, hasn't expired, and has `passed` set to `true`.

Perform a protected action only when the last statement holds.
