At the end of this guide, your app can tell which of your registered Bluetooth beacons a device is near, and Radar generates an event when the device comes into range of a beacon and when it leaves. A detection means the device is within radio range of that beacon: it is proximity, not an indoor coordinate.

## Complete example

The app initializes, identifies the user, requests permission, ranges beacons in a `trackOnce()` call, and reads the beacons in range from the result and from entry and exit events.

<CodeGroup>

```swift Swift
import CoreLocation
import RadarSDK
import UIKit

@main
class AppDelegate: UIResponder, UIApplicationDelegate, RadarDelegate {
    private let locationManager = CLLocationManager()

    func application(
        _ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?
    ) -> Bool {
        Radar.initialize(publishableKey: "prj_test_pk_...")
        Radar.setUserId("user-123")
        Radar.setDelegate(self)
        return true
    }

    // Call when the user reaches the indoor feature. Ranging needs location access.
    func requestLocationPermission() {
        locationManager.requestWhenInUseAuthorization()
    }

    // Call where indoor presence matters, for example when a store-mode screen opens.
    func checkNearbyBeacons() {
        Radar.trackOnce(desiredAccuracy: .high, beacons: true) { status, location, events, user in
            guard status == .success, let user = user else {
                // No result. Show the geofence-level experience.
                print("trackOnce failed: \(Radar.stringForStatus(status))")
                return
            }

            let beacons = user.beacons ?? []
            if beacons.contains(where: { $0.tag == "store-register" }) {
                print("At a register")
            } else {
                print("No register beacon in range")
            }
        }
    }

    func didReceiveEvents(_ events: [RadarEvent], user: RadarUser?) {
        for event in events {
            switch event.type {
            case .userEnteredBeacon:
                print("Entered beacon \(event.beacon?.externalId ?? "")")
            case .userExitedBeacon:
                print("Exited beacon \(event.beacon?.externalId ?? "")")
            default:
                break
            }
        }
    }

    func didFail(status: RadarStatus) {
        print("Radar request failed: \(Radar.stringForStatus(status))")
    }
}
```

```kotlin Kotlin
import android.Manifest
import android.app.Application
import android.content.Context
import android.location.Location
import android.os.Build
import android.util.Log
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import io.radar.sdk.Radar
import io.radar.sdk.RadarReceiver
import io.radar.sdk.RadarTrackingOptions
import io.radar.sdk.model.RadarEvent
import io.radar.sdk.model.RadarUser

class MyApplication : Application() {
    override fun onCreate() {
        super.onCreate()
        Radar.initialize(this, "prj_test_pk_...")
        Radar.setUserId("user-123")
        Radar.setReceiver(BeaconReceiver())
    }
}

class MainActivity : AppCompatActivity() {
    private val permissionRequest = registerForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { grants ->
        if (grants.values.all { it }) checkNearbyBeacons()
    }

    // Call when the user reaches the indoor feature. Ranging needs precise
    // location and, on Android 12 and later, Bluetooth access.
    fun requestBeaconPermissions() {
        val permissions = mutableListOf(Manifest.permission.ACCESS_FINE_LOCATION)
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            permissions += Manifest.permission.BLUETOOTH_SCAN
            permissions += Manifest.permission.BLUETOOTH_CONNECT
        }
        permissionRequest.launch(permissions.toTypedArray())
    }

    // Call where indoor presence matters, for example when a store-mode screen opens.
    fun checkNearbyBeacons() {
        Radar.trackOnce(
            RadarTrackingOptions.RadarTrackingOptionsDesiredAccuracy.HIGH,
            true
        ) { status, location, events, user ->
            if (status != Radar.RadarStatus.SUCCESS || user == null) {
                // No result. Show the geofence-level experience.
                Log.w("Beacons", "trackOnce failed: $status")
                return@trackOnce
            }

            val beacons = user.beacons ?: emptyArray()
            if (beacons.any { it.tag == "store-register" }) {
                Log.d("Beacons", "At a register")
            } else {
                Log.d("Beacons", "No register beacon in range")
            }
        }
    }
}

class BeaconReceiver : RadarReceiver() {
    override fun onEventsReceived(context: Context, events: Array<RadarEvent>, user: RadarUser?) {
        for (event in events) {
            when (event.type) {
                RadarEvent.RadarEventType.USER_ENTERED_BEACON ->
                    Log.d("Beacons", "Entered beacon ${event.beacon?.externalId}")
                RadarEvent.RadarEventType.USER_EXITED_BEACON ->
                    Log.d("Beacons", "Exited beacon ${event.beacon?.externalId}")
                else -> Unit
            }
        }
    }

    override fun onError(context: Context, status: Radar.RadarStatus) {
        Log.w("Beacons", "Radar request failed: $status")
    }

    override fun onLocationUpdated(context: Context, location: Location, user: RadarUser) {}

    override fun onClientLocationUpdated(
        context: Context,
        location: Location,
        stopped: Boolean,
        source: Radar.RadarLocationSource
    ) {}

    override fun onLog(context: Context, message: String) {}
}
```

```javascript React Native
  console.warn('Radar request failed', status);

// Call when the user reaches the indoor feature. Ranging needs location
// access and, on Android 12 and later, Bluetooth access.
export async function requestBeaconPermissions() {
  const location = await Radar.requestPermissions(false);
  if (location !== 'GRANTED_FOREGROUND' && location !== 'GRANTED_BACKGROUND') {
    return false;
  }
  if (Platform.OS === 'android' && Platform.Version >= 31) {
    const grants = await PermissionsAndroid.requestMultiple([
      PermissionsAndroid.PERMISSIONS.BLUETOOTH_SCAN,
      PermissionsAndroid.PERMISSIONS.BLUETOOTH_CONNECT,
    ]);
    return Object.values(grants).every((grant) => grant === PermissionsAndroid.RESULTS.GRANTED);
  }
  return true;
}

// Call where indoor presence matters, for example when a store-mode screen opens.
export async function checkNearbyBeacons() {
  try {
    const { user } = await Radar.trackOnce({ desiredAccuracy: 'high', beacons: true });
    const beacons = user?.beacons ?? [];
    if (beacons.some((beacon) => beacon.tag === 'store-register')) {
      console.log('At a register');
    } else {
      console.log('No register beacon in range');
    }
  } catch (err) {
    // No result. Show the geofence-level experience.
    console.warn('trackOnce failed', err.code);
  }
}
```

## Prerequisites

- A Radar project with Beacons, which depends on your plan.
- The [iOS](/sdk/ios), [Android](/sdk/android), or [React Native](/sdk/react-native) SDK, initialized, with the user identified. The web SDK doesn't range beacons. Ranging on Android needs Android 8.0 (API level 26) or later.
- Beacon hardware that broadcasts iBeacon. Android can also range Eddystone-UID. iOS can't.
- A physical device with Bluetooth. Simulators and emulators can't range beacons.

## Create beacons

Register each beacon so that the SDK knows to listen for it. A beacon is identified by a `tag` (a group, such as `store-register`) and an `externalId` (your ID for the unit), and is created separately in Test and Live. Creating a beacon with a `tag` and `externalId` that already exist updates it, which is how you replace a failed unit.

Each beacon also needs the identifiers it broadcasts (`uuid`, `major`, and `minor` for `ibeacon`, or `uid` and `instance` for `eddystone`) and approximate `coordinates` as `[longitude,latitude]`. Radar uses the coordinates only to choose which beacons to send to a device: up to 10 within 1 kilometer of its location. Use the building's location.

Configure each beacon with its vendor's tool first, and give each physical unit a unique combination of identifiers. The detection area is set at the beacon, by its transmit power and where you mount it, so start with low power and raise it.

- **Dashboard**: On the [Beacons](https://dashboard.radar.com/geofencing/beacons) page, click **New**, enter the details, and click **Create**.
- **CSV import**: On the Beacons page, click **Import** and select a CSV file with a header row. The columns are `description`, `tag`, `externalId`, `type`, `uuid`, `major`, `minor`, `uid`, `instance`, `coordinates`, `enabled`, and `metadata`.
- **API**: Call [Upsert a beacon](/api/beacons#upsert-a-beacon) from your backend, as in the complete example.

If a site has more than 10 beacons that share a UUID, enter the UUID under **iBeacon UUIDs** in the same section. The SDK then ranges every beacon that broadcasts it, regardless of registered coordinates. On iOS, this also needs the **Location updates** background mode.

## Request permissions

Ranging needs location permission on both platforms: foreground access for `trackOnce()`, and background access for background monitoring. See [Understand foreground and background location permissions](/fundamentals/location-permissions). On Android, it needs precise location (`ACCESS_FINE_LOCATION`), because an approximate-only grant isn't enough, and these Bluetooth permissions in your manifest:

```xml AndroidManifest.xml
<manifest xmlns:android="http://schemas.android.com/apk/res/android">
    <uses-permission android:name="android.permission.BLUETOOTH" />
    <uses-permission android:name="android.permission.BLUETOOTH_ADMIN" />
    <uses-permission android:name="android.permission.BLUETOOTH_CONNECT" />
    <uses-permission android:name="android.permission.BLUETOOTH_SCAN" />
</manifest>
```

If your app targets API level 31 or later, `BLUETOOTH_SCAN` and `BLUETOOTH_CONNECT` are runtime permissions. Request them before you range.

## Detect beacons in the foreground

Call `trackOnce()` with `beacons` set to `true` and high desired accuracy at the point where indoor presence matters, for example when the user opens a store-mode screen. The SDK gets the device's location, asks Radar for registered beacons nearby, listens for them for up to about 5 seconds, and sends what it heard. The call takes a few seconds longer than a `trackOnce()` without beacons, so show a loading state.

## Monitor beacons in the background

To generate beacon events while the app isn't open, turn on `beacons` in your tracking options and start tracking after the user grants background location permission:

<CodeGroup>

```swift Swift
let trackingOptions = RadarTrackingOptions.presetResponsive
trackingOptions.beacons = true
Radar.startTracking(trackingOptions: trackingOptions)
```

```kotlin Kotlin
import io.radar.sdk.RadarTrackingOptions

val trackingOptions = RadarTrackingOptions.RESPONSIVE.copy(beacons = true)
Radar.startTracking(trackingOptions)
```

</CodeGroup>

Background detection is opportunistic and can be delayed, like background location. Use foreground ranging when the user needs an immediate result.

## Read the returned data

`user.beacons` lists the registered, enabled beacons heard during the most recent update. Branch on `tag` and `externalId`. A device in range of several beacons gets all of them, so decide what overlap means for your feature. An empty list means no registered beacon was heard during that update. It doesn't prove that the user is away from every beacon.

In the native SDKs, each beacon also has `rssi`, the signal strength. Use it to compare beacons while you tune. Don't convert it to a distance.

| Event | When it fires |
| --- | --- |
| `user.entered_beacon` | A beacon is heard that wasn't in the user's previous state. |
| `user.exited_beacon` | A beacon in the user's previous state is no longer heard. |

Handle these in the same delegate or receiver as geofence events, and on your backend by [webhook](/integrations/webhooks). Use events for one-time triggers and `user.beacons` for what the user sees now.

To include beacons in a verified location check, pass `beacons: true` to `Radar.trackVerified()`, and make the decision on your backend from the validated result. See the [Geo-compliance gaming guide](/geo-compliance/gaming).

## Handle results and errors

| Result | Meaning | What to do |
| --- | --- | --- |
| A Bluetooth error (`.errorBluetooth`, `ERROR_BLUETOOTH`) in `didFail(status:)` or `onError()` | Bluetooth is off or unavailable. | Ask the user to turn on Bluetooth, and fall back to the geofence-level experience. |
| A permissions error in `onError()` on Android, while `trackOnce()` reports success | Bluetooth permission isn't granted. The update is sent without beacons. | Check the permission before you call. Don't read a result with no beacons as "not near a beacon". |
| `trackOnce()` fails with a permissions status | Location permission is denied. | Show the default experience. |
| An expected beacon is missing | It isn't broadcasting, its identifiers or environment don't match, it is disabled, or its registered coordinates are more than 1 kilometer from the device's reported location. | Confirm the broadcast with a Bluetooth scanner, then compare the identifiers, environment, and coordinates in Radar. |
| An Eddystone beacon is missing on iOS | iOS ranges iBeacon only. | Use iBeacon hardware. |
| A beacon is heard outside its intended area | Transmit power is too high. Beacons are often heard through walls and floors. | Lower the transmit power at the beacon, then walk the boundary with a test device. |
| Alternating entry and exit events | The device is at the edge of the beacon's range. | Adjust power or placement. Don't trigger irreversible actions on a single exit event. |

## Test the integration

Test on a physical device, in the Test environment.

- **Presence**: Stand next to a registered beacon and run the foreground check. `user.beacons` contains the beacon with your tag and external ID, and `user.entered_beacon` appears on the [Events](https://dashboard.radar.com/geofencing/events) page.
- **Absence**: Move well out of range, or power the beacon off, and run the check again. `user.beacons` is empty, and `user.exited_beacon` appears on the Events page.

See [Test, troubleshoot, and go live](/fundamentals/testing-and-launch).

## Production checklist

- [ ] Every installed beacon is registered in Live with a unique `tag` and `externalId`, and the physical unit is labeled with the `externalId`.
- [ ] Transmit power and placement were tuned on site, including checks through walls and on adjacent floors.
- [ ] Beacon event types are turned on in Live.
- [ ] Android manifest entries and runtime Bluetooth permission requests are in place.
- [ ] The app has a defined behavior for no beacons, several beacons, Bluetooth off, and permission denied.
- [ ] No business rule treats "no beacon detected" as proof of absence.
