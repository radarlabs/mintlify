At the end of this page, your app can open inside one of your locations and show the right experience immediately, without waiting for an entry event. Geofencing tells your app and your backend where a user is in terms you define: which of your locations they are in, which places they are visiting, how far they are from a destination, and which beacons are nearby.

## Quickstart

This quickstart needs only foreground location permission and the Test environment. The code initializes the SDK, identifies the user, requests permission, and reads the user's current geofences each time the app becomes active.

<CodeGroup>

```swift Swift
import CoreLocation
import RadarSDK
import UIKit

@main
class AppDelegate: UIResponder, UIApplicationDelegate {
    private let locationManager = CLLocationManager()

    func application(
        _ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?
    ) -> Bool {
        Radar.initialize(publishableKey: "prj_test_pk_...")
        Radar.setUserId("user-123")

        // Runs at launch, on resume, and after the user answers the permission prompt.
        NotificationCenter.default.addObserver(
            forName: UIApplication.didBecomeActiveNotification, object: nil, queue: .main
        ) { [weak self] _ in
            self?.refreshLocationContext()
        }
        return true
    }

    // Call when the user reaches a location feature.
    func requestLocationPermission() {
        locationManager.requestWhenInUseAuthorization()
    }

    func refreshLocationContext() {
        Radar.trackOnce { status, location, events, user in
            guard status == .success, let user = user else {
                // No location context. Show the default experience.
                print("trackOnce failed: \(Radar.stringForStatus(status))")
                return
            }

            if let store = user.geofences?.first(where: { $0.tag == "store" }) {
                print("In store \(store.externalId ?? "")")
            } else {
                print("Not in a store")
            }
        }
    }
}
```

```kotlin Kotlin
import android.Manifest
import android.app.Application
import android.util.Log
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import io.radar.sdk.Radar

class MyApplication : Application() {
    override fun onCreate() {
        super.onCreate()
        Radar.initialize(this, "prj_test_pk_...")
        Radar.setUserId("user-123")
    }
}

class MainActivity : AppCompatActivity() {
    private val permissionRequest = registerForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { grants ->
        if (grants.values.any { it }) refreshLocationContext()
    }

    // Call when the user reaches a location feature.
    fun requestLocationPermission() {
        permissionRequest.launch(
            arrayOf(
                Manifest.permission.ACCESS_FINE_LOCATION,
                Manifest.permission.ACCESS_COARSE_LOCATION
            )
        )
    }

    override fun onStart() {
        super.onStart()
        refreshLocationContext()
    }

    private fun refreshLocationContext() {
        Radar.trackOnce { status, location, events, user ->
            if (status != Radar.RadarStatus.SUCCESS || user == null) {
                // No location context. Show the default experience.
                Log.w("Geofencing", "trackOnce failed: $status")
                return@trackOnce
            }

            val store = user.geofences?.firstOrNull { it.tag == "store" }
            if (store != null) {
                Log.d("Geofencing", "In store ${store.externalId}")
            } else {
                Log.d("Geofencing", "Not in a store")
            }
        }
    }
}
```

```javascript JavaScript
import Radar from 'radar-sdk-js';

Radar.initialize('prj_test_pk_...');
Radar.setUserId('user-123');

// The browser asks for location permission on the first call.
async function refreshLocationContext() {
  try {
    const { user } = await Radar.trackOnce();
    const store = user?.geofences?.find((geofence) => geofence.tag === 'store');
    if (store) {
      console.log(`In store ${store.externalId}`);
    } else {
      console.log('Not in a store');
    }
  } catch (err) {
    // No location context. Show the default experience.
    console.warn('trackOnce failed', err.status);
  }
}

refreshLocationContext();
document.addEventListener('visibilitychange', () => {
  if (document.visibilityState === 'visible') refreshLocationContext();
});
```

```dart Flutter
import 'package:flutter/widgets.dart';
import 'package:flutter_radar/flutter_radar.dart';

// Call from main(), after WidgetsFlutterBinding.ensureInitialized().
Future<void> setUpRadar() async {
  await Radar.initialize('prj_test_pk_...');
  await Radar.setUserId('user-123');
  WidgetsBinding.instance.addObserver(_ResumeObserver());
  await refreshLocationContext();
}

// Call when the user reaches a location feature.
Future<void> requestLocationPermission() async {
  await Radar.requestPermissions(false);
  final status = await Radar.getPermissionsStatus();
  if (status == 'GRANTED_FOREGROUND' || status == 'GRANTED_BACKGROUND') {
    await refreshLocationContext();
  }
}

Future<void> refreshLocationContext() async {
  final res = await Radar.trackOnce();
  if (res?['status'] != 'SUCCESS') {
    // No location context. Show the default experience.
    debugPrint('trackOnce failed: ${res?['status'] ?? res?['error']}');
    return;
  }

  final geofences = (res?['user']?['geofences'] as List?) ?? [];
  final stores = geofences.where((geofence) => geofence['tag'] == 'store');
  if (stores.isNotEmpty) {
    debugPrint('In store ${stores.first['externalId']}');
  } else {
    debugPrint('Not in a store');
  }
}

class _ResumeObserver with WidgetsBindingObserver {
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) refreshLocationContext();
  }
}
```

<Steps>
  <Step title="Get your Test publishable key">
    Copy it from the [Settings](https://dashboard.radar.com/settings) page. See [Choose the right API key](/fundamentals/authentication-and-access#choose-the-right-api-key).
  </Step>
  <Step title="Create a geofence around a place you can test">
    On the [Geofences](https://dashboard.radar.com/geofencing/geofences) page, with the environment set to **Test**, click **Create**, search for your current address, and select **circle**. Enter the tag `store` and the external ID `store-42`, then click **Create**. Make the circle large enough to contain the spot where you test, because indoors a phone's reported location can be tens of meters off.
  </Step>
  <Step title="Install and initialize the SDK">
    Follow the install steps for [iOS](/sdk/ios), [Android](/sdk/android), [web](/sdk/web), [React Native](/sdk/react-native), or [Flutter](/sdk/flutter), and initialize once, at app startup, with the Test publishable key.
  </Step>
  <Step title="Run the app inside the geofence">
    Grant location permission when the app asks. `trackOnce()` returns the user with your geofence in `user.geofences`.
  </Step>
</Steps>

The [Users](https://dashboard.radar.com/geofencing/users) page shows your device, and the [Events](https://dashboard.radar.com/geofencing/events) page shows a `user.entered_geofence` event. If `geofences` is empty, check that the geofence and the publishable key are in the same environment, and that the device's reported location falls inside the circle. See [Test, troubleshoot, and go live](/fundamentals/testing-and-launch).

## Read current state or wait for an event

Read current state (`user.geofences`, `user.place`, `user.beacons`, and `user.trip` from `trackOnce()`) for what the user sees on screen, because it is true on every update while the user is inside, including when the app opens with the user already there. Use events, such as `user.entered_geofence`, for one-time triggers such as a notification or a backend workflow, because Radar generates an event once, at the transition. Events while the app is closed need background location permission and `Radar.startTracking()`.

## Custom geofences and Places

| | Custom geofences | Places |
| --- | --- | --- |
| What they represent | Locations you define: your stores, venues, or job sites. | Places from Radar's dataset, grouped by chain (such as `starbucks`) and category (such as `airport`). |
| How you set them up | Create or import them. | Turn on Places and choose chains and categories. |
| User context | `user.geofences`, a list. | `user.place`, one place. |
| Events | `user.entered_geofence`, `user.exited_geofence`, `user.dwelled_in_geofence` | `user.entered_place`, `user.exited_place` |

Use custom geofences for locations you operate, and Places for locations you don't, such as partner venues or airports.
