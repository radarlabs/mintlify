```html JavaScript
      const form = document.getElementById('address-form');
      const statusEl = document.getElementById('status');
      const choicesEl = document.getElementById('choices');
      const fields = {};
      ['line1', 'unit', 'city', 'stateCode', 'postalCode', 'countryCode'].forEach((id) => {
        fields[id] = document.getElementById(id);
      });

      // The last result taken from Radar. Cleared when the user edits a field by hand.
      let accepted = null;

      Radar.initialize('prj_test_pk_...');

      function setStatus(message) {
        statusEl.textContent = message;
      }

      // Copy a Radar address result into the editable fields.
      function fillFields(address) {
        fields.line1.value = [address.number, address.street].filter(Boolean).join(' ') || address.addressLabel || '';
        fields.unit.value = address.unit || fields.unit.value;
        fields.city.value = address.city || '';
        fields.stateCode.value = address.stateCode || '';
        fields.postalCode.value = address.postalCode || '';
        fields.countryCode.value = address.countryCode || '';
        accepted = address;
      }

      // Editing anything except the unit means the fields no longer match a Radar result.
      ['line1', 'city', 'stateCode', 'postalCode', 'countryCode'].forEach((id) => {
        fields[id].addEventListener('input', () => {
          accepted = null;
        });
      });

      function submitAddress(address) {
        choicesEl.replaceChildren();
        setStatus(`Address accepted: ${address.formattedAddress}`);
        // Send the field values, address.latitude, and address.longitude to your backend here.
      }

      Radar.ui.autocomplete({
        container: 'autocomplete',
        width: '100%',
        layers: ['address'],
        countryCode: 'US',
        placeholder: 'Start typing your address',
        ariaLabel: 'Find your address',
        onSelection: (address) => {
          fillFields(address);
          choicesEl.replaceChildren();
          setStatus('Check the details, add a unit if you have one, then continue.');
        },
        onError: (error) => {
          console.error(error);
          setStatus('Address suggestions are unavailable. Enter your address in the fields below.');
        },
      });

      form.addEventListener('submit', async (event) => {
        event.preventDefault();
        choicesEl.replaceChildren();

        const missing = ['line1', 'city', 'postalCode', 'countryCode']
          .filter((id) => !fields[id].value.trim());
        if (missing.length > 0) {
          setStatus('Enter a street address, city, postal code, and country.');
          return;
        }

        // The fields still match a result the user chose. No second request is needed.
        if (accepted) {
          submitAddress(accepted);
          return;
        }

        // The address was typed or edited by hand. Check that it resolves.
        const query = [
          fields.line1.value.trim(),
          fields.city.value.trim(),
          `${fields.stateCode.value.trim()} ${fields.postalCode.value.trim()}`.trim(),
        ].join(', ');

        setStatus('Checking address...');

        let addresses;
        try {
          ({ addresses } = await Radar.forwardGeocode({
            query,
            layers: ['address'],
            country: fields.countryCode.value.trim(),
          }));
        } catch (error) {
          // API error: bad request, rate limit, network failure, or server error.
          console.error(error);
          setStatus(error.status === 'ERROR_BAD_REQUEST'
            ? 'That address could not be read. Check it and try again.'
            : 'We could not check the address right now. Try again.');
          return;
        }

        // No result.
        if (!addresses || addresses.length === 0) {
          setStatus('We could not find that address. Check the street number, street name, and postal code.');
          return;
        }

        // One exact match: a valid address.
        if (addresses.length === 1 && addresses[0].confidence === 'exact') {
          fillFields(addresses[0]);
          submitAddress(addresses[0]);
          return;
        }

        // Several matches, or an approximate match: ask the user to choose or correct.
        setStatus('Is one of these your address? Choose one, or edit the fields and try again.');
        addresses.slice(0, 5).forEach((address) => {
          const button = document.createElement('button');
          button.type = 'button';
          button.textContent = address.formattedAddress;
          button.addEventListener('click', () => {
            fillFields(address);
            submitAddress(address);
          });
          choicesEl.append(button);
        });
      });
```

```swift Swift
import UIKit

@main
class AppDelegate: UIResponder, UIApplicationDelegate {
    func application(
        _ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?

func suggestAddresses(for query: String) {
    Radar.autocomplete(query: query, near: nil, layers: ["address"], limit: 8, country: "US") { (status, addresses) in
        guard status == .success, let addresses = addresses else {
            // Show that suggestions are unavailable. Manual entry still works.
            return
        }

        // Show each address's formattedAddress as a suggestion, or a "no results" row when the array is empty.
        print(addresses.count)
    }
}

// Call when the user chooses a suggestion, to fill the editable fields.
func fillFields(from address: RadarAddress) {
    let numberAndStreet = [address.number, address.street].compactMap { $0 }.joined(separator: " ")
    let line1 = numberAndStreet.isEmpty ? (address.addressLabel ?? "") : numberAndStreet
    let unit = address.unit ?? ""
    let city = address.city ?? ""
    let stateCode = address.stateCode ?? ""
    let postalCode = address.postalCode ?? ""
    let countryCode = address.countryCode ?? ""

    // Set your text fields from these values, and keep address.coordinate with them.
    print(line1, unit, city, stateCode, postalCode, countryCode)
}

// Call on submit when the address was typed or edited by hand.
func checkTypedAddress(line1: String, city: String, stateCode: String, postalCode: String, countryCode: String) {
    let query = "\(line1), \(city), \(stateCode) \(postalCode)"

    Radar.geocode(address: query, layers: ["address"], countries: [countryCode]) { (status, addresses) in
        guard status == .success, let addresses = addresses else {
            // Show a retry message.
            return
        }

        if addresses.isEmpty {
            // No result. Ask the user to check the street number, street name, and postal code.
        } else if addresses.count == 1 && addresses[0].confidence == .exact {
            // One exact match: a valid address.
            fillFields(from: addresses[0])
        } else {
            // Several matches, or an approximate match. Show up to five and ask the user to choose or correct.
            print(addresses.prefix(5).compactMap { $0.formattedAddress })
        }
    }
}
```

```jsx React Native
import { useState } from 'react';
import { Button, Text, TextInput, View } from 'react-native';
import Radar, { Autocomplete } from 'react-native-radar';

Radar.initialize('prj_test_pk_...');

const LABELS = {
  line1: 'Street address',
  unit: 'Apartment, suite, or unit',
  city: 'City',
  stateCode: 'State',
  postalCode: 'Postal code',
  countryCode: 'Country',
};

// Copy a Radar address result into the editable fields.
const toFields = (address, unit) => ({
  line1: [address.number, address.street].filter(Boolean).join(' ') || address.addressLabel || '',
  unit: address.unit || unit,
  city: address.city || '',
  stateCode: address.stateCode || '',
  postalCode: address.postalCode || '',
  countryCode: address.countryCode || '',
});

export default function AddressForm({ onAccept }) {
  const [fields, setFields] = useState({ line1: '', unit: '', city: '', stateCode: '', postalCode: '', countryCode: 'US' });
  // The last result taken from Radar. Cleared when the user edits a field by hand.
  const [accepted, setAccepted] = useState(null);
  const [choices, setChoices] = useState([]);
  const [status, setStatus] = useState('');

  const fill = (address) => {
    const next = toFields(address, fields.unit);
    setFields(next);
    setAccepted(address);
    setChoices([]);
    return next;
  };

  const edit = (key, value) => {
    setFields({ ...fields, [key]: value });
    if (key !== 'unit') setAccepted(null);
  };

  const submit = async () => {
    setChoices([]);
    if (!fields.line1.trim() || !fields.city.trim() || !fields.postalCode.trim() || !fields.countryCode.trim()) {
      setStatus('Enter a street address, city, postal code, and country.');
      return;
    }

    // The fields still match a result the user chose. No second request is needed.
    if (accepted) {
      onAccept(fields, accepted);
      return;
    }

    // The address was typed or edited by hand. Check that it resolves.
    setStatus('Checking address...');
    let addresses;
    try {
      ({ addresses } = await Radar.geocode({
        address: `${fields.line1}, ${fields.city}, ${fields.stateCode} ${fields.postalCode}`,
        layers: ['address'],
        countries: [fields.countryCode],
      }));
    } catch (err) {
      setStatus('We could not check the address right now. Try again.');
      return;
    }

    if (!addresses || addresses.length === 0) {
      setStatus('We could not find that address. Check the street number, street name, and postal code.');
    } else if (addresses.length === 1 && addresses[0].confidence === 'exact') {
      setStatus('');
      onAccept(fill(addresses[0]), addresses[0]);
    } else {
      // Several matches, or an approximate match: ask the user to choose or correct.
      setStatus('Is one of these your address? Choose one, or edit the fields and try again.');
      setChoices(addresses.slice(0, 5));
    }
  };

  return (
    <View>
      <Autocomplete
        options={{
          layers: ['address'],
          countryCode: 'US',
          placeholder: 'Start typing your address',
          onSelection: (address) => {
            fill(address);
            setStatus('Check the details, add a unit if you have one, then continue.');
          },
          onError: () => setStatus('Address suggestions are unavailable. Enter your address in the fields below.'),
        }}
      />
      {Object.keys(LABELS).map((key) => (
        <TextInput
          key={key}
          accessibilityLabel={LABELS[key]}
          placeholder={LABELS[key]}
          value={fields[key]}
          onChangeText={(value) => edit(key, value)}
        />
      ))}
      <Button title="Continue" onPress={submit} />
      <Text accessibilityLiveRegion="polite">{status}</Text>
      {choices.map((address) => (
        <Button
          key={address.formattedAddress}
          title={address.formattedAddress}
          onPress={() => onAccept(fill(address), address)}
        />
      ))}
    </View>
  );
}
```

## Prerequisites

- A Radar account and a publishable key. See [Keys](#keys).
- **Web**: The examples use `radar-sdk-js` 5.3.0 and `@radarlabs/plugin-autocomplete` 1.2.0. The autocomplete plugin requires `radar-sdk-js` 5.3.0 or later. The geocoding and validation methods are part of the core SDK and need no plugin.
- **Permissions**: None for autocomplete, forward geocoding, IP geocoding, or validation. Reverse geocoding the user's current position needs location permission. Reverse geocoding a coordinate you already have doesn't.
- **Plan**: Address validation is available on the [Enterprise plan](https://radar.com/pricing), for United States and Canada addresses. Address enrichment on autocomplete selections must be enabled for your project. Everything else in this guide is available to any project.

## How it works

### Five operations

These are separate operations with different inputs and outputs. Choose by what you have and what you need.

| Operation | You have | You get | Use it for | Reference |
| --- | --- | --- | --- | --- |
| Autocomplete | A partial address or place name, as the user types | A ranked list of candidate addresses | Address entry in forms and search boxes | [Autocomplete](/api/search#autocomplete) |
| Forward geocode | A complete address as text | Matching addresses with coordinates and a confidence level | Turning a stored or typed address into a coordinate | [Forward geocode](/api/geocoding#forward-geocode) |
| Reverse geocode | A coordinate | The address or region at that coordinate | Labeling a map pin or a device position | [Reverse geocode](/api/geocoding#reverse-geocode) |
| IP geocode | Nothing. Radar reads the request's IP address | An estimated city, state, and country | Defaults such as country or language, with no permission prompt | [IP geocode](/api/geocoding#ip-geocode) |
| Validate | A complete address in structured fields | A standardized address and a verification status | Checking an address before you ship or bill to it | [Validate an address](/api/geocoding#validate-an-address) |

Autocomplete is for partial input and forward geocoding is for complete addresses. Don't call forward geocoding on every keystroke, and don't use autocomplete to resolve a stored address in a batch job.

### Formatted address, geocoded coordinate, and validated address

A Radar address result can carry three different kinds of information. They answer different questions.

| Term | Where it appears | What it tells you | What it doesn't tell you |
| --- | --- | --- | --- |
| Formatted address | `formattedAddress` on every address result | A display string assembled from the result's fields, such as `841 Broadway, New York, NY 10003 USA`. | That the address was checked. A formatted string exists for a city or postal code result too. |
| Geocoded coordinate | `latitude`, `longitude`, `layer`, and, on forward geocode results, `confidence` | Where Radar places the result, and how precisely. `exact` is a known address location, `interpolated` is an estimate between known address points, and `fallback` is the center of a containing area such as a postal code or city. | That mail or a courier can reach it. A `fallback` coordinate can be far from the building. |
| Validated address | The `address` and `result.verificationStatus` returned by the validate operation | Whether the structured input matched a known address record, in standardized form with fields such as `plus4` and a record type. | That a delivery will succeed, that the recipient is there, or that an unverified address doesn't exist. |

A coordinate tells you where to put a pin. A verification status tells you whether the text of the address matched a record. Don't treat either as the other.

### Responsibilities

| Part | Responsibility |
| --- | --- |
| Your form | Renders the autocomplete input and editable fields, applies the selected result, checks hand-typed input on submit, and shows correction messages. |
| Radar | Returns candidates, coordinates, estimates, and verification results. |
| Your backend | Receives the final fields and coordinate, and applies your own rules, such as whether to accept an unverified address. If the decision matters, repeat the check on your backend, because anything sent from a browser can be changed by the user. |

`container` is an element ID or an element. If you pass an existing `input` element, the component attaches its suggestions list to that input and leaves the input's styling alone. If the container can't be found, `Radar.ui.autocomplete()` throws an error with the `status` `CONTAINER_NOT_FOUND`.

To verify, type three or more characters. Suggestions appear under the input, the arrow keys move through them, and Enter selects one.

The JavaScript tab in [Complete example](#complete-example) combines the component with editable fields and a check on submit. To run it, save it as an HTML file, replace the key, and open it in a browser. The sections that follow explain each part.

### Fill structured fields from the selection

`onSelection` receives one address object. The fields most forms need are:

| Field | Example | Notes |
| --- | --- | --- |
| `number` | `841` | The street number. |
| `street` | `Broadway` | The street name. |
| `unit` | `Fl 7` | Present when the result identifies a unit. |
| `addressLabel` | `841 Broadway` | The first line of the address. |
| `city` | `New York` | |
| `stateCode` | `NY` | `state` has the full name. |
| `postalCode` | `10003` | |
| `countryCode` | `US` | `country` has the full name. |
| `formattedAddress` | `841 Broadway, New York, NY 10003 USA` | For display. |
| `latitude`, `longitude` | `40.7343`, `-73.9911` | Also available as a GeoJSON point in `geometry`, in `[longitude, latitude]` order. |
| `layer` | `address` | The kind of result. A `place` result also has `placeLabel`. |

Every field except the coordinate can be absent, so read them defensively. A result from the `postalCode` or `locality` layer has no street number, which is why the example limits suggestions to `layers: ['address']` and still leaves every field editable.

Store the structured fields and the coordinate, not only the formatted string. For the full list of address fields, see [Address object](/api/geocoding#address-object).

### Let the user correct the result

Autocomplete suggests an address. It doesn't decide which address is right. Keep the fields editable after a selection so the user can add a unit, fix a postal code, or enter an address Radar doesn't know, such as a new building.

The example tracks whether the fields still match a Radar result with the `accepted` variable. Adding a unit keeps the match. Editing any other field clears it, so the address is checked again on submit.

### Check a typed address on submit

When the user typed or edited the address by hand, the example sends the combined fields to `Radar.forwardGeocode()` with `layers: ['address']` and the country, and branches on the response:

| Response | Meaning | What the example does |
| --- | --- | --- |
| One address with `confidence` of `exact` | The text resolved to one known address location. | Accepts it and updates the fields to the matched form. |
| Several addresses, or one with `confidence` of `interpolated` or `fallback` | The input is ambiguous, or Radar could only place it approximately. | Shows up to five candidates as buttons and lets the user choose one or edit the fields. |
| An empty `addresses` array | Nothing matched. | Asks the user to check the street number, street name, and postal code. Doesn't submit. |
| The promise rejects | The request failed. | Shows a retry message. |

Whether your product blocks submission when no address matches is your decision. A checkout for physical delivery might require a match or an explicit "use this address anyway" confirmation. A contact form might accept anything. Make that rule explicit, and enforce it on your backend as well.

This check confirms that the address resolves to a location. It isn't address validation. If you need a verification status and a standardized postal form, see [Validate an address](#validate-an-address).

These options control when the component searches and what it returns.

| `mailable` | none | Return only mailable addresses. |
| `placeholder` | `Search address` | Placeholder text for the input. |
| `width` | `400` | Width in pixels or as a CSS value. With `responsive` set to `true` (the default) it acts as a maximum width. |
| `maxHeight` | none | Maximum height of the suggestions list. |
| `disabled` | `false` | Disables the input. |
| `showMarkers` | `true` | Shows a marker icon next to each suggestion. |

### Use your own input

To keep an existing input and suggestions list, call the autocomplete operation yourself through an `AutocompleteSession`. Create one session per address field. The session groups the field's requests and lets you report the result the user chose. This example adds a minimum input length and a debounce:

```javascript
const session = new Radar.AutocompleteSession();
const input = document.getElementById('address-input');
let timer;

input.addEventListener('input', () => {
  clearTimeout(timer);
  const query = input.value.trim();

  // Minimum input: wait for three characters.
  if (query.length < 3) {
    renderSuggestions([]);
    return;
  }

  // Debounce: wait 200 ms after the last keystroke.
  timer = setTimeout(async () => {
    try {
      const { addresses } = await session.autocomplete({
        query,
        layers: ['address'],
        countryCode: 'US',
        limit: 5,
      });

      // Ignore a response for text the user has already changed.
      if (input.value.trim() === query) {
        renderSuggestions(addresses);
      }
    } catch (error) {
      console.error(error);
      showSuggestionsUnavailable();
    }
  }, 200);
});

// Call this when the user chooses a suggestion.
function chooseSuggestion(address) {
  fillFields(address);
  session.select(address).catch(() => {});
}
```

`renderSuggestions`, `showSuggestionsUnavailable`, and `fillFields` are your own functions. `session.select()` reports the chosen result to Radar, which uses selections to measure and improve autocomplete quality. Pass it the address object exactly as `session.autocomplete()` returned it. If the project has address enrichment enabled, the promise resolves with an enriched `address`. The component does all of this for you.

## Use the other operations

### Forward geocode

Convert a complete address to a coordinate, as in this example:

```javascript
try {
  const { addresses } = await Radar.forwardGeocode({ query: '841 Broadway, New York, NY 10003' });

  if (addresses.length === 0) {
    // No match.
  } else {
    const { latitude, longitude, confidence, formattedAddress } = addresses[0];
  }
} catch (error) {
  console.error(error.status, error.message);
}
```

Optional parameters are `layers`, `country`, and `lang`. Check `confidence` before you rely on the coordinate for anything precise, such as a delivery pin: `exact` and `interpolated` are address-level, and `fallback` is the center of a larger area. See [Forward geocode](/api/geocoding#forward-geocode).

### Reverse geocode

Convert a coordinate to an address, as in this example:

```javascript
const { addresses } = await Radar.reverseGeocode({ latitude: 40.7343, longitude: -73.9911 });

if (addresses.length > 0) {
  console.log(addresses[0].formattedAddress);
}
```

Pass both `latitude` and `longitude`. If you omit them, the SDK requests the browser's location first, which shows a permission prompt and can reject with `ERROR_PERMISSIONS` or `ERROR_LOCATION`. Use `layers` to choose the level of detail, for example `['locality']` when you only need the city. The result is the address nearest the coordinate. It isn't proof that a person or device was inside that building. See [Reverse geocode](/api/geocoding#reverse-geocode).

### Validate an address

<Info>
  Address validation is available on the [Enterprise plan](https://radar.com/pricing), for United States and Canada addresses.
</Info>

Validation takes a complete address in structured fields and returns a standardized address and a verification status. A project without access receives a `402` response, which the web SDK reports as `ERROR_PAYMENT_REQUIRED`.

Send `city`, `stateCode`, `postalCode`, and `countryCode`, plus either `number` and `street` or a single `addressLabel`. Sending `addressLabel` together with `number` and `street` returns a `400` error.

This example validates an address and branches on the verification status:

```javascript
async function validate(fields) {
  try {
    const { address, result } = await Radar.validateAddress(fields);

    switch (result && result.verificationStatus) {
      case 'verified':
        // The input matched one address record completely. Use the standardized address.
        return { ok: true, address };
      case 'partially verified':
        // Part of the input matched. Show the standardized address and ask the user to confirm it.
        return { ok: false, suggestion: address };
      case 'ambiguous':
        // The input closely matches several records. Ask for the missing detail, often a unit.
        return { ok: false, needsDetail: true };
      default:
        // 'unverified', or no status: Radar could not verify the input.
        return { ok: false };
    }
  } catch (error) {
    // Validation was unavailable. Decide whether your flow continues without it.
    console.error(error.status, error.message);
    return { ok: false, unavailable: true };
  }
}

const outcome = await validate({
  number: '841',
  street: 'Broadway',
  unit: 'Fl 7',
  city: 'New York',
  stateCode: 'NY',
  postalCode: '10003',
  countryCode: 'US',
});
```

| `verificationStatus` | Meaning | Suggested handling |
| --- | --- | --- |
| `verified` | The input has a complete and unique match with a result. | Accept. Store the standardized `address`. |
| `partially verified` | The input has a partial match with a result. | Show the returned address next to what the user typed and ask them to confirm or correct. |
| `ambiguous` | The input closely matches multiple results. | Ask for more detail. |
| `unverified` | Radar was unable to verify the input address. | Ask the user to correct it, or let them continue with an explicit confirmation if your product allows that. |

A `verified` status means the text matched a single known address record. It isn't a guarantee that a carrier delivers there or that the recipient lives there, and an `unverified` status doesn't prove the address is fake. New construction and rural addresses are common reasons for an address to be real but unverified. For United States addresses, the response can also carry an analysis of the unit, indicating that a unit is missing or invalid. The standardized `address` includes `plus4` and `metadata` with a record type and property type. For the full response, see [Validate an address](/api/geocoding#validate-an-address).

### IP geocode

IP geocoding converts the IP address of the request to an estimated city, state, and country. It needs no permission and no input. This example returns region defaults, or none when there's no estimate or the request came through a proxy:

```javascript
async function regionDefaults() {
  try {
    const { address, proxy } = await Radar.ipGeocode();

    if (!address || proxy) {
      // No estimate, or the request came through a known proxy. Do not assume a region.
      return { countryCode: null, stateCode: null };
    }

    return { countryCode: address.countryCode, stateCode: address.stateCode };
  } catch (error) {
    return { countryCode: null, stateCode: null };
  }
}
```

The response has `ip`, `address` (with `countryCode`, `stateCode`, `city`, `postalCode`, and a coarse `latitude` and `longitude`), and `proxy`, which is `true` when the IP address is a known proxy.

Treat the result as an estimate. An IP address locates a network, not a person. Mobile carriers, corporate networks, and VPNs routinely place a user in a different city, state, or country. Good uses are defaults the user can change:

- Preselect the country in an address form and pass it as `countryCode` to autocomplete.
- Choose an initial language, currency, or regional content.
- Pick which regional notice or consent option to show first during onboarding. When there is no estimate or `proxy` is `true`, fall back to your most protective default. An IP estimate is one input to that choice. It doesn't establish which laws apply to a user and isn't a consent solution on its own.

Don't use IP geocoding to prove that a user is physically in a place or to decide eligibility for regulated activity. For that, see [Geo-compliance overview and quickstart](/geo-compliance/overview). For the response fields, see [IP geocode](/api/geocoding#ip-geocode).

| --- | --- |

| `Radar.reverseGeocode()` | `addresses`: an array of address objects. |

### Errors, ambiguous results, and recovery

Web SDK methods reject with a `RadarError`. Read its `status` and `message`. The component passes the same errors to `onError`.

| Situation | What happens | What to do |
| --- | --- | --- |
| Fewer characters than `minCharacters` | The component sends no request and shows no list. | Nothing. |
| No suggestions for the text | The component's list shows "No results". A direct call resolves with an empty `addresses` array. | Let the user keep typing or enter the address in the fields by hand. |
| The user never picks a suggestion | `onSelection` doesn't fire. | Keep the fields usable on their own, and check them on submit. |
| Several matches for a typed address | Forward geocoding returns more than one address. | Show the candidates and let the user choose. Don't pick the first one silently for a delivery address. |
| Approximate match | `confidence` is `interpolated` or `fallback`, or `layer` isn't `address`. | Ask the user to confirm, or ask for a more complete address. |
| No match | An empty `addresses` array. | Ask for a correction. Decide whether your flow allows continuing. |
| Invalid request | `ERROR_BAD_REQUEST`, for example an empty query or conflicting validation fields. | Fix the request. Don't retry unchanged. |
| Wrong or revoked key | `ERROR_UNAUTHORIZED`. | Check the key and environment. |
| Validation not available to the project | `ERROR_PAYMENT_REQUIRED`. | Fall back to the forward geocode check, or contact your Radar account team about access. |
| Rate limited | `ERROR_RATE_LIMIT`. | Keep the debounce and minimum length in place, and retry after a short wait. |
| Network failure or timeout | `ERROR_NETWORK` or another `RadarError`. | Let the user retry, and keep manual entry available. |
| Location denied during reverse geocoding with no coordinate | `ERROR_PERMISSIONS`. | Ask for an address instead. |
| The component's container is missing | `Radar.ui.autocomplete()` throws with `status` `CONTAINER_NOT_FOUND`. | Create the component after the container is in the document. |
| The component is recreated on every render | Duplicate inputs or lost focus in a single-page app. | Create it once per mount and call `remove()` on unmount. |

Address suggestions are an aid. If Radar is unreachable, the user must still be able to type an address into the fields and submit it under whatever rule your product sets.

## Test the integration

1. **Valid address from a suggestion**: Type the start of a real address, choose a suggestion, and confirm every field is filled and the status reads "Check the details". Select Continue. The address is accepted with no further request.
2. **Valid address typed by hand**: Without using the suggestions, type a complete real address into the fields and select Continue. One exact match is accepted, and the fields update to the matched form.
3. **Ambiguous input**: Enter an address that Radar can only match approximately, for example a real street and city with a street number that doesn't exist on that street. Candidate buttons appear. Choose one and confirm it is accepted. If your input returns a single exact match, it wasn't ambiguous, so try another. In an automated test, stub `Radar.forwardGeocode` to resolve with two addresses.
4. **No result**: Enter a street line of random letters and select Continue. The correction message appears and nothing is accepted.
5. **API error**: In developer tools, block requests to `api.radar.io`. Type in the search input and confirm the "suggestions are unavailable" message. Then fill the fields by hand and select Continue, and confirm the "could not check" message. Unblock and confirm both paths recover.
6. **Edit after selecting**: Choose a suggestion, change the postal code, and select Continue. The address is checked again and isn't accepted as-is.
7. **Unit**: Choose a suggestion, add a unit, and select Continue. The address is accepted and the unit is kept.
8. **Keyboard and screen reader**: Reach the search input with Tab, move through suggestions with the arrow keys, select with Enter, and close the list with Escape. A screen reader announces the number of results.
9. **Validation, if you use it**: Send one address you expect to be `verified`, one with a missing unit in a multi-unit building, and one that doesn't exist, and confirm your handling of each status.
10. **IP defaults, if you use them**: Load the page through a VPN and confirm that the defaults are either reasonable or absent and that the user can change them.

## Production checklist

- [ ] The form uses the Live publishable key, supplied by configuration. No secret key is in client code.
- [ ] SDK and plugin versions are pinned.
- [ ] Minimum input length and debounce are in place for every autocomplete input, including inputs you built yourself.
- [ ] `layers` and `countryCode` match the addresses you accept.
- [ ] Fields stay editable after a selection, and manual entry works when suggestions are unavailable.
- [ ] You store structured fields and the coordinate, not only `formattedAddress`.
- [ ] Ambiguous, approximate, no-match, and error cases each have a message and a next step.
- [ ] Your rule for unverified or unmatched addresses is written down and enforced on the backend.
- [ ] Wording in your interface doesn't promise more than the result supports: "verified" isn't "deliverable".
- [ ] IP-derived defaults are labeled or changeable, and aren't used for eligibility decisions.
- [ ] Addresses and coordinates are personal data. Don't write them to client logs or analytics in production.
- [ ] You have checked [Coverage](#coverage) for the countries you serve. You can cache API responses for up to 30 days under the [terms of use](https://radar.com/terms).
