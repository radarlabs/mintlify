## Complete example

Each tab renders a Radar map centered on Union Square in New York, adds one marker, and shows a message if the map fails to load, and the sections below explain each part.

<CodeGroup>

```html JavaScript
<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <link href="https://js.radar.com/maps/v1.2.0/radar-maps.css" rel="stylesheet">
    <script src="https://js.radar.com/v5.3.0/radar.min.js"></script>
    <script src="https://js.radar.com/maps/v1.2.0/radar-maps.min.js"></script>
  </head>

  <body>
    <p id="map-status" role="status">Loading map...</p>
    <div id="map" style="width: 100%; height: 500px;"></div>

    <script>
      const statusEl = document.getElementById('map-status');
      let loaded = false;

      try {
        Radar.initialize('prj_test_pk_...');

        const map = Radar.ui.map({
          container: 'map',
          style: 'radar-default-v1',
          center: [-73.9905, 40.7359], // [longitude, latitude]
          zoom: 14,
        });

        map.on('load', () => {
          loaded = true;
          statusEl.hidden = true;

          Radar.ui.marker({ popup: { text: 'Union Square' } })
            .setLngLat([-73.9905, 40.7359])
            .addTo(map);
        });

        map.on('error', (event) => {
          console.error('Map error', event.error);
          if (!loaded) {
            statusEl.textContent = 'The map could not load. Refresh the page to try again.';
          }
        });
      } catch (err) {
        // Thrown when the key is missing or is a secret key, or when the plugin did not load.
        console.error(err);
        statusEl.textContent = 'The map could not load.';
      }
    </script>
  </body>
</html>
```

```jsx React
import { useEffect, useRef, useState } from 'react';
import Radar from 'radar-sdk-js';
import { createMapsPlugin } from '@radarlabs/plugin-maps';
import '@radarlabs/plugin-maps/dist/radar-maps.css';

Radar.registerPlugin(createMapsPlugin());
Radar.initialize(process.env.REACT_APP_RADAR_PUBLISHABLE_KEY);

const CENTER = [-73.9905, 40.7359]; // [longitude, latitude]

export default function RadarMap() {
  const containerRef = useRef(null);
  const [status, setStatus] = useState('loading');

  useEffect(() => {
    let loaded = false;

    const map = Radar.ui.map({
      container: containerRef.current,
      style: 'radar-default-v1',
      center: CENTER,
      zoom: 14,
    });

    map.on('load', () => {
      loaded = true;
      setStatus('ready');

      Radar.ui.marker({ popup: { text: 'Union Square' } })
        .setLngLat(CENTER)
        .addTo(map);
    });

    map.on('error', (event) => {
      console.error('Map error', event.error);
      if (!loaded) {
        setStatus('error');
      }
    });

    // Remove the map when the component unmounts.
    return () => map.remove();
  }, []);

  return (
    <div>
      {status === 'loading' && <p role="status">Loading map...</p>}
      {status === 'error' && <p role="alert">The map could not load. Refresh the page to try again.</p>}
      <div ref={containerRef} style={{ width: '100%', height: '500px' }} />
    </div>
  );
}
```

```swift Swift
import SwiftUI
import MapLibre

struct RadarMap: UIViewRepresentable {
    let publishableKey = "prj_test_pk_..."
    let center = CLLocationCoordinate2D(latitude: 40.7359, longitude: -73.9905)
    var onLoadFailed: (Error) -> Void

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    func makeUIView(context: Context) -> MLNMapView {
        let styleURL = URL(string: "https://api.radar.io/maps/styles/radar-default-v1?publishableKey=\(publishableKey)")

        let mapView = MLNMapView(frame: .zero, styleURL: styleURL)
        mapView.autoresizingMask = [.flexibleWidth, .flexibleHeight]
        mapView.setCenter(center, zoomLevel: 14, animated: false)
        mapView.delegate = context.coordinator

        // The Radar logo replaces the MapLibre logo. Add radar-logo to your asset catalog.
        mapView.logoView.isHidden = true
        let logoImageView = UIImageView(image: UIImage(named: "radar-logo"))
        logoImageView.translatesAutoresizingMaskIntoConstraints = false
        mapView.addSubview(logoImageView)
        NSLayoutConstraint.activate([
            logoImageView.bottomAnchor.constraint(equalTo: mapView.safeAreaLayoutGuide.bottomAnchor, constant: -10),
            logoImageView.leadingAnchor.constraint(equalTo: mapView.safeAreaLayoutGuide.leadingAnchor, constant: 10),
            logoImageView.widthAnchor.constraint(equalToConstant: 74),
            logoImageView.heightAnchor.constraint(equalToConstant: 26)
        ])

        return mapView
    }

    func updateUIView(_ uiView: MLNMapView, context: Context) {
    }

    class Coordinator: NSObject, MLNMapViewDelegate {
        let control: RadarMap

        init(_ control: RadarMap) {
            self.control = control
        }

        // Add the marker when the style finishes loading. Add default-marker to your asset catalog.
        func mapView(_ mapView: MLNMapView, didFinishLoading style: MLNStyle) {
            let point = MLNPointAnnotation()
            point.coordinate = control.center

            let shapeSource = MLNShapeSource(identifier: "marker-source", shape: point, options: nil)
            let shapeLayer = MLNSymbolStyleLayer(identifier: "marker-style", source: shapeSource)

            if let image = UIImage(named: "default-marker") {
                style.setImage(image, forName: "marker")
            }
            shapeLayer.iconImageName = NSExpression(forConstantValue: "marker")

            style.addSource(shapeSource)
            style.addLayer(shapeLayer)
        }

        func mapViewDidFailLoadingMap(_ mapView: MLNMapView, withError error: Error) {
            control.onLoadFailed(error)
        }
    }
}
```

## Prerequisites

- A Radar account and a publishable key. See [Keys](#keys).
- **Web**: A browser with WebGL. The examples use `radar-sdk-js` 5.3.0 and `@radarlabs/plugin-maps` 1.2.0. The maps plugin requires `radar-sdk-js` 5.1.0 or later and `maplibre-gl` as a peer dependency (2.4, 3.x, 4.x, 5.0.1 or later, or 6.4.1 or later).
- **Native**: MapLibre Native for iOS or Android, the Radar React Native SDK with `@maplibre/maplibre-react-native` 11, or `maplibre_gl` for Flutter. See [Compatibility](#compatibility).
- **Permissions**: None to display a map. You need location permission only if you also show the user's position.
- **Plan**: Satellite and hybrid styles are available on the [Enterprise plan](https://radar.com/pricing).

### Keys

<EnvironmentAndKeys />

Map display uses a publishable key only. The web SDK sends it as a request header. Style URLs that you give to another map library, and static map URLs, carry it in a `publishableKey` query parameter.

## How it works

| Part | Responsibility |
| --- | --- |
| Your page or app | Loads the SDK and stylesheet, provides a sized container, creates the map, and adds markers and shapes from your data. |
| Radar web SDK and maps plugin | Resolves the Radar style, attaches your publishable key to style and tile requests, adds the Radar logo, attribution, and zoom controls, and exposes `Radar.ui.map()`, `Radar.ui.marker()`, and `Radar.ui.popup()`. |
| Radar | Serves the style, vector tiles, fonts, and hosted marker images. |
| Your backend | Supplies the locations to show. It isn't involved in rendering the map and doesn't need a Radar key for this guide. |

`Radar.ui.map()` returns a `RadarMap`, which extends the MapLibre GL JS [`Map`](https://maplibre.org/maplibre-gl-js/docs/API/classes/Map/) class. Every MapLibre map method and event is available on it. The MapLibre module itself is available as `Radar.ui.maplibregl`.

## Getting started

Add the Radar SDK and the maps plugin to your app. In web SDK 5 and later, maps are a separate plugin, and `Radar.ui` doesn't exist until the plugin is registered.

### With npm or yarn

Install the core SDK, the maps plugin, and MapLibre GL JS:

```bash
npm install radar-sdk-js @radarlabs/plugin-maps maplibre-gl
```

Import the plugin stylesheet, register the plugin, then initialize with your publishable key:

```javascript
import Radar from 'radar-sdk-js';
import { createMapsPlugin } from '@radarlabs/plugin-maps';
import '@radarlabs/plugin-maps/dist/radar-maps.css';

Radar.registerPlugin(createMapsPlugin());

Radar.initialize('prj_test_pk_...');
```

The stylesheet import is required. It includes the MapLibre styles, so you don't import MapLibre's CSS separately. Without it the map canvas, controls, markers, and popups are mispositioned.

### With HTML

Add the stylesheet, the core SDK, and the maps plugin, in that order. The plugin script registers itself with the core SDK and bundles MapLibre GL JS.

```html
<link href="https://js.radar.com/maps/v1.2.0/radar-maps.css" rel="stylesheet">
<script src="https://js.radar.com/v5.3.0/radar.min.js"></script>
<script src="https://js.radar.com/maps/v1.2.0/radar-maps.min.js"></script>
```

Then initialize the SDK:

```html
<script>
  Radar.initialize('prj_test_pk_...');
</script>
```

### Create a map

Give the map a container with an explicit width and height, then pass the container's ID or the element itself to `Radar.ui.map()`. A container with no height renders nothing.

The examples below load three locations, show a loading message until the map's `load` event, add a marker and popup for each location, fit the view to the markers, and show a message if the map fails to load.

## Configuration

### Map

`Radar.ui.map()` accepts any MapLibre [MapOptions](https://maplibre.org/maplibre-gl-js/docs/API/type-aliases/MapOptions/) except `transformRequest`, which the SDK uses to attach your key. These are the options most maps set:

| Option | Type | Default | Description |
| --- | --- | --- | --- |
| `container` | string or `HTMLElement` | None | Required. The element ID or element to render into. Its width and height set the map size. |
| `style` | string or object | `radar-default-v1` | The map style: a Radar style name, a custom style ID, a style URL, or a MapLibre style object. See [Styles](#styles). |
| `center` | `[longitude, latitude]` | Estimated from the client's IP address | The initial center. If you omit it, Radar centers the map on an estimate of the client's location from its IP address. That estimate is approximate. |
| `zoom` | number | Set by the style | The initial zoom level. Higher numbers are closer. |
| `language` | string | None | The language for map labels: a language code such as `en` or `es`, or `local` to use the browser's language. |
| `showZoomControls` | boolean | `true` | Whether to show the zoom in and zoom out buttons. |

The SDK also changes four MapLibre defaults: `maxZoom` is `20`, `dragRotate` is `false`, `touchPitch` is `false`, and MapLibre's own attribution control and logo are replaced by Radar's. Pass any of the first three options to override them.

This example sets the most common options:

```javascript
const map = Radar.ui.map({
  container: 'map',
  style: 'radar-default-v1',
  center: [-73.990550, 40.735225],
  zoom: 10,
  language: 'en',
});
```

#### Functions

`RadarMap` adds these methods to the MapLibre map:

| Function | Arguments | Returns | Description |
| --- | --- | --- | --- |
| `getMarkers` | none | `RadarMarker[]` | All markers currently on the map. |
| `fitToMarkers` | `fitBoundsOptions` (optional), `RadarMarker[]` (optional) | void | Fits the view to all markers, or to the markers you pass. Default padding is 50 pixels. Does nothing when there are no markers. |
| `clearMarkers` | none | void | Removes all markers. |
| `getFeatures` | none | `RadarMapFeature[]` | All features added with the methods below. |
| `fitToFeatures` | `fitBoundsOptions` (optional), `RadarMapFeature[]` (optional) | void | Fits the view to all features, or to the features you pass. |
| `clearFeatures` | none | void | Removes all features. |
| `addPolygon` | GeoJSON `Polygon` or `MultiPolygon` feature, options (optional) | `RadarMapFeature` | Adds a [polygon](#polygon-and-multipolygon). |
| `addLine` | GeoJSON `LineString` feature, options (optional) | `RadarMapFeature` | Adds a [line](#line-and-polyline). |
| `addPolyline` | encoded polyline string, options (optional) | `RadarMapFeature` | Decodes a polyline and adds it as a line. Precision defaults to 6. |

`fitBoundsOptions` are MapLibre [FitBoundsOptions](https://maplibre.org/maplibre-gl-js/docs/API/type-aliases/FitBoundsOptions/), such as `padding` and `maxZoom`. Use `map.on()` for MapLibre [map events](https://maplibre.org/maplibre-gl-js/docs/API/type-aliases/MapEventType/):

```javascript
map.on('load', () => {
  // The style has loaded and the map is ready.
});

map.on('click', (event) => {
  console.log(event.lngLat);
});

map.fitToMarkers({ padding: 20 });
map.fitToFeatures({ padding: 20 });
```

### Styles

Radar provides these styles:

| Style | Description |
| --- | --- |
| `radar-default-v1` | Radar's default style, with street-level detail for roads and nearby places. A general-purpose map for most uses. |
| `radar-light-v1` | A light monochrome style for highlighting your own data. |
| `radar-dark-v1` | A dark monochrome style for highlighting your own data in dark interfaces. |
| `radar-satellite-v1` | Satellite imagery. Available on the [Enterprise plan](https://radar.com/pricing). |
| `radar-hybrid-v1` | Satellite imagery with roads and labels. Available on the Enterprise plan. |

To switch styles after the map is created, call MapLibre's `map.setStyle()` with a Radar style name or custom style ID.

#### Custom styles

Radar Maps Studio is a visual style editor in the dashboard. Use it to match the map to your brand.

1. In the dashboard, open **Maps** > **Styles** and click **Create style**. Choose a base style and click **Create**.
2. In the editor, select a layer such as background, roads, or water to show or hide it or change its color.
3. To add a custom marker image, click **+** next to **Add a marker** in the layer panel and upload an SVG or PNG file. Use the marker's ID with the `marker` option described in [Marker](#marker).
4. For more control, turn on **Advanced mode**. It exposes every layer, lets you reorder layers, adds attributes such as opacity, border color, text color, and fonts, and lets you edit the style JSON, which follows the [MapLibre Style Spec](https://maplibre.org/maplibre-style-spec).
5. To use your own font, select the **Settings** tab, click **Upload fonts**, and upload a TTF or OTF file. The font then appears in the **Text font** list. Custom fonts require Advanced mode.
6. Click **Save**. For a new style, the confirmation dialog shows the style ID. You can also find the ID in the style's **share** dialog. Changes to a style that is already in use are visible to users as soon as you save and confirm.

Pass the style ID as the `style` option:

```javascript
const map = Radar.ui.map({
  container: 'map',
  style: 'CUSTOM_STYLE_ID',
});
```

### Marker

`Radar.ui.marker()` returns a `RadarMarker`, which extends the MapLibre [`Marker`](https://maplibre.org/maplibre-gl-js/docs/API/classes/Marker/) class. Position it with `setLngLat([longitude, latitude])` and add it with `addTo(map)`.

| Option | Type | Default | Description |
| --- | --- | --- | --- |
| `color` | string | `#000257` | Color of the default marker. |
| `marker` | string | None | ID of a marker image uploaded to a [custom style](#custom-styles). |
| `url` | string | None | URL of an image to use as the marker. |
| `element` | `HTMLElement` | None | A DOM element to use as the marker. |
| `width` | number or string | None | Image width, in pixels or as a CSS value. |
| `height` | number or string | None | Image height, in pixels or as a CSS value. |
| `scale` | number | `1` | Scale of the default marker. |
| `popup` | object | None | [Popup](#popup) options for a popup that opens when the marker is selected. |

When you set `url` or `marker` without a width or height, the image is capped at 64 by 64 pixels. If the image fails to load, the SDK logs an error and shows the default marker.

Only `color`, `element`, and `scale` are passed through from MapLibre's marker options. For other MapLibre marker behavior, call the marker's methods after you create it, for example `marker.setDraggable(true)`.

This example adds a marker with a popup and listens for clicks:

```javascript
const marker = Radar.ui.marker({
  color: '#000257',
  popup: {
    text: 'Flatiron',
  },
})
  .setLngLat([-73.9897, 40.7411])
  .addTo(map);

marker.on('click', (event) => {
  console.log(event.lngLat, event.target);
});
```

Selecting a marker opens its popup and closes any other open marker popup. The marker's `click` event receives `target` (the marker), `lngLat`, `point`, and `originalEvent`. A click on a marker doesn't also fire the map's `click` event.

### Popup

A popup shows text or HTML at a point on the map. Attach one to a marker with the marker's `popup` option, or create a standalone popup with `Radar.ui.popup()`. Popups accept all MapLibre [PopupOptions](https://maplibre.org/maplibre-gl-js/docs/API/type-aliases/PopupOptions/) plus:

| Option | Type | Default | Description |
| --- | --- | --- | --- |
| `text` | string | None | Plain text to show in the popup. |
| `html` | string | None | An HTML string to show in the popup. |
| `element` | `HTMLElement` | None | A DOM element to use as the popup content. |

This example creates a standalone popup:

```javascript
const popup = Radar.ui.popup({
  text: 'Flatiron',
})
  .setLngLat([-73.9897, 40.7411])
  .addTo(map);
```

<Warning>
  `html` is inserted as markup. If any part of it comes from user input or from data you don't control, use `text` or build an `element` with `textContent` to avoid script injection.
</Warning>

### Features

Features are lines and polygons that you add to the map, style, and listen to. The input is a [GeoJSON Feature](https://datatracker.ietf.org/doc/html/rfc7946#section-3.2) with `Polygon`, `MultiPolygon`, or `LineString` geometry.

- Each feature needs a unique `id`. Adding a second feature with the same ID throws an error. If you omit the ID, the SDK generates one.
- `properties` carry your own data and are returned in feature events.
- `feature.on(type, callback)` supports `click`, `mousemove`, `mouseenter`, and `mouseleave`. The callback receives an object with `type`, `feature`, and `originalEvent` (the MapLibre mouse event, which has `lngLat`).
- `feature.remove()` removes the feature and its layers.

#### Polygon and MultiPolygon

Add polygons with `map.addPolygon(feature, options)`. The `paint` option supports:

| Property | Possible values | Default |
| --- | --- | --- |
| `fill-color` | string (CSS color) | `#FF6F00` |
| `fill-opacity` | number (0 to 1) | `0.3` |
| `border-width` | number (pixels) | `2` |
| `border-color` | string (CSS color) | `#FF6F00` |
| `border-opacity` | number (0 to 1) | `1` |

This example adds a polygon and opens a popup when the user selects it:

```javascript
const geojson = {
  type: 'Feature',
  id: 'service-area-1',
  properties: {
    name: 'Central Park',
  },
  geometry: {
    type: 'Polygon',
    coordinates: [[
      [-73.95820498349376, 40.80026256420331],
      [-73.9816545434309, 40.76824212209061],
      [-73.97283590550576, 40.764244806438285],
      [-73.94918592197929, 40.79677293396304],
      [-73.95820498349376, 40.80026256420331],
    ]],
  },
};

const feature = map.addPolygon(geojson, {
  paint: {
    'fill-color': 'blue',
    'fill-opacity': 0.4,
  },
});

feature.on('click', ({ feature, originalEvent }) => {
  Radar.ui.popup({ text: feature.properties.name })
    .setLngLat(originalEvent.lngLat)
    .addTo(map);
});
```

#### Line and Polyline

Add a GeoJSON `LineString` with `map.addLine(feature, options)`, or an encoded polyline with `map.addPolyline(polyline, options)`. For polylines, `options` also takes `id`, `properties`, and `precision` (default `6`; use `5` for polylines encoded with five decimal places). The `paint` option supports:

| Property | Possible values | Default |
| --- | --- | --- |
| `line-color` | string (CSS color) | `#000257` |
| `line-width` | number (pixels) | `4` |
| `line-opacity` | number (0 to 1) | None |
| `line-cap` | `butt`, `round`, or `square` | `round` |
| `line-offset` | number (pixels) | None |
| `line-blur` | number (pixels) | None |
| `line-dasharray` | array of numbers | None |
| `line-gap-width` | number (pixels) | None |
| `line-gradient` | MapLibre [expression](https://maplibre.org/maplibre-gl-js/docs/examples/line-gradient/) | None |
| `border-width` | number (pixels) | `2` |
| `border-color` | string (CSS color) | `#FFFFFF` |
| `border-opacity` | number (0 to 1) | None |

This example adds an encoded polyline as a styled line:

```javascript
const feature = map.addPolyline(encodedPolyline, {
  id: 'route-1',
  precision: 6,
  properties: {
    name: 'Delivery route',
  },
  paint: {
    'line-color': 'blue',
    'line-opacity': 0.6,
    'line-width': 6,
    'border-width': 2,
    'border-color': '#FFFFFF',
  },
});
```

## Examples

Each example assumes the SDK is initialized and `map` was created as in [Create a map](#create-a-map).

### Fit map to markers

`map.fitToMarkers()` centers and zooms the map so every marker is visible. This example adds a marker where the user clicks, refits the map, and removes a marker when it is selected.

```javascript
map.on('click', (event) => {
  const { lng, lat } = event.lngLat;

  const marker = Radar.ui.marker()
    .setLngLat([lng, lat])
    .addTo(map);

  map.fitToMarkers({ maxZoom: 14, padding: 80 });

  marker.on('click', () => {
    marker.remove();
    map.fitToMarkers({ maxZoom: 14, padding: 80 });
  });
});
```

### Add a custom icon

Pass an image URL as the marker's `url`, or the ID of a marker uploaded to a custom style as `marker`.

```javascript
map.on('load', () => {
```

### Display a custom popup

Build popup content as a DOM element when it includes your own data. Setting `textContent` keeps location names and addresses from being read as markup.

```javascript
function buildPopup(location) {
  const element = document.createElement('div');

  const title = document.createElement('strong');
  title.textContent = location.name;

  const address = document.createElement('p');
  address.textContent = location.address;

  element.append(title, address);
  return element;
}

Radar.ui.marker({
  popup: {
    element: buildPopup({ name: 'Flatiron', address: '123 Example Ave, New York, NY' }),
    maxWidth: '260px',
  },
})
  .setLngLat([-73.9897, 40.7411])
  .addTo(map);
```

### Display a route

Drawing a route is optional. To show one, request it with geometry and add the geometry as a line.

`Radar.distance()` returns each requested travel mode's route. With `geometry: 'linestring'`, the route's `geometry` is a GeoJSON `LineString` that you can pass to `map.addLine()`, as this example does:

```javascript
async function showRoute(map, origin, destination) {
  const { routes } = await Radar.distance({
    origin,       // { latitude, longitude }
    destination,  // { latitude, longitude }
    modes: ['car'],
    units: 'imperial',
    geometry: 'linestring',
  });

  if (!routes.car || !routes.car.geometry) {
    return; // No drivable route between the two points.
  }

  map.clearFeatures();

  map.addLine({
    type: 'Feature',
    id: 'route',
    properties: { duration: routes.car.duration.text },
    geometry: routes.car.geometry,
  }, {
    paint: { 'line-color': '#000257', 'line-width': 5 },
  });

  map.fitToFeatures({ padding: 40 });
}
```

The [directions](/api/routing#directions) endpoint returns encoded polylines for routes with several stops. Request `geometry=polyline6` and pass each polyline to `map.addPolyline()`, whose default precision is 6. For live trip routes and ETAs, see [Track pickups, deliveries, and workers](/geofencing/trips).

### Display a polygon

Pass a GeoJSON `Polygon` or `MultiPolygon` feature to `map.addPolygon()`, then fit the view to it.

```javascript
map.on('load', () => {

    id: 'central-park',
    properties: { name: 'Central Park' },
    geometry: {

        [-73.95820498349376, 40.80026256420331],
```

Because `RadarMap` is a MapLibre map, MapLibre-compatible drawing plugins work with it. If you let users draw shapes and save them as geofences, send the finished geometry to your own backend and create the geofence there with a secret key. See [Detect geofence events and place visits](/geofencing/geofences-and-places) and the [Geofences API](/api/geofences).

## Static maps

A static map is an image that Radar renders from a URL. Use one where you can't or don't want to run JavaScript or a map library: emails, receipts, order confirmations, and list thumbnails. A static map doesn't pan or zoom.

Build the URL from the endpoint `https://api.radar.io/maps/static` and query parameters, then use it as the source of an image element. The key goes in the `publishableKey` query parameter, as in this URL:

```text
https://api.radar.io/maps/static?width=600&height=400&center=40.73430,-73.99110&zoom=15&style=radar-default-v1&scale=2&markers=color:0x000257|40.73430,-73.99110&publishableKey=prj_live_pk_...
```

This returns a 600 by 400 point image at double resolution, centered on the coordinate, with one marker. Give the image element alternative text that describes the place shown.

| Parameter | Default | Description |
| --- | --- | --- |
| `width`, `height` | None | Required. Image size in pixels, each between `100` and `2048`. The size multiplied by `scale` must not exceed `2048`. |
| `center` | None | `latitude,longitude`. Required unless `markers`, `path`, or `bbox` is set. |
| `zoom` | None | Required when `center` is set. If you omit `center` and `zoom`, the map fits the markers, paths, or bounding box. |
| `scale` | `1` | A number from `1` to `3`. Use `2` or `3` for high-density screens. The returned image is `width` and `height` multiplied by `scale`, covering the same area. |
| `style` | `radar-default-v1` | A Radar style name or [custom style](#custom-styles) ID. |
| `bbox` | None | `min_latitude,min_longitude,max_latitude,max_longitude`. The image fits the box. |
| `language` | None | A language code for labels. |
| `markers` | None | One or more markers. See [Static map markers](#static-map-markers). |
| `path` | None | One or more lines. See [Static map paths](#static-map-paths). |
| `publishableKey` | None | Required. Your publishable key. |

For every parameter and error, see [Maps API](/api/maps).

### Static map markers

A `markers` value is a set of optional style settings followed by one or more `latitude,longitude` locations, separated by pipes. For example:

```text
markers=color:0x000257|size:small|40.73430,-73.99110|40.706436,-73.997271
```

Style settings are `color` (a hex value such as `0xFF428E` or a CSS color name), `size:small` for a smaller marker, and `icon` (the URL of a PNG or SVG file, or the ID of a marker uploaded to a custom style; images are capped at 64 by 64 pixels). To use more than one marker style in an image, repeat the `markers` parameter.

### Static map paths

A `path` value is a set of optional style settings followed by `latitude,longitude` points or an encoded polyline. Points come after the style settings. For example:

```text
path=stroke:0x000257|width:3|border:white|borderwidth:2|40.7154,-74.0147|40.7094,-74.0149|40.7049,-74.0167
```

Style settings are `stroke` (line color), `width` (pixels, default `3`), `border` (border color, default white), and `borderwidth` (pixels, default `2`). To pass an encoded polyline, use `enc:` followed by the polyline, or `enc6:` for a polyline with six decimal places. Radar [routing](/api/routing) responses can return route geometry as a polyline, so you can draw a route on a static map without decoding it. Repeat the `path` parameter for several lines, and URL-encode polylines before you place them in a URL.

## Compatibility

Radar maps use the [Mapbox Vector Tile](https://github.com/mapbox/vector-tile-spec) and [TileJSON](https://github.com/mapbox/tilejson-spec) standards and MapLibre styles, so any library that supports MapLibre styles can display them. Outside the Radar web SDK, you load a style by URL, with your publishable key in the query string:

```text
https://api.radar.io/maps/styles/radar-default-v1?publishableKey=prj_live_pk_...
```

Replace `radar-default-v1` with another [style](#styles) name or a custom style ID.

When you use a library directly, you're responsible for two things the Radar web SDK otherwise does for you: keep the library's attribution control visible, and add the Radar logo to the map. See [Attribution](#attribution).

### iOS

Add [MapLibre Native for iOS](https://maplibre.org/maplibre-native/ios/latest/documentation/maplibre/) to your Xcode project, then create an `MLNMapView` with a Radar style URL. Add the Radar logo as an image asset named `radar-logo` (and optionally a marker image named `default-marker`). Logo and marker assets are in the [Radar map assets download](https://static.radar.com/downloads/radar-map-assets-v1.zip).

This SwiftUI example creates the map view, adds the Radar logo, and adds a marker when the style loads:

```swift
import SwiftUI
import MapLibre

struct MapView: UIViewRepresentable {
    func makeCoordinator() -> MapView.Coordinator {
        Coordinator(self)
    }

    func makeUIView(context: Context) -> MLNMapView {
        // Create a map with a Radar style.
        let style = "radar-default-v1"
        let publishableKey = "prj_live_pk_..."
        let styleURL = URL(string: "https://api.radar.io/maps/styles/\(style)?publishableKey=\(publishableKey)")

        let mapView = MLNMapView(frame: .zero, styleURL: styleURL)
        mapView.autoresizingMask = [.flexibleWidth, .flexibleHeight]
        mapView.logoView.isHidden = true // Replaced by the Radar logo.

        mapView.setCenter(
            CLLocationCoordinate2D(latitude: 40.7342, longitude: -73.9911),
            zoomLevel: 11,
            animated: false
        )

        // Add the Radar logo.
        let logoImageView = UIImageView(image: UIImage(named: "radar-logo"))
        logoImageView.translatesAutoresizingMaskIntoConstraints = false
        mapView.addSubview(logoImageView)

        NSLayoutConstraint.activate([
            logoImageView.bottomAnchor.constraint(equalTo: mapView.safeAreaLayoutGuide.bottomAnchor, constant: -10),
            logoImageView.leadingAnchor.constraint(equalTo: mapView.safeAreaLayoutGuide.leadingAnchor, constant: 10),
            logoImageView.widthAnchor.constraint(equalToConstant: 74),
            logoImageView.heightAnchor.constraint(equalToConstant: 26)
        ])

        mapView.delegate = context.coordinator

        return mapView
    }

    func updateUIView(_ uiView: MLNMapView, context: Context) {
    }

    class Coordinator: NSObject, MLNMapViewDelegate {
        var control: MapView

        init(_ control: MapView) {
            self.control = control
        }

        // Add a marker when the style finishes loading.
        func mapView(_ mapView: MLNMapView, didFinishLoading style: MLNStyle) {
            addMarker(style: style, coordinate: CLLocationCoordinate2D(latitude: 40.7342, longitude: -73.9911))
        }

        func addMarker(style: MLNStyle, coordinate: CLLocationCoordinate2D) {
            let point = MLNPointAnnotation()
            point.coordinate = coordinate

            let shapeSource = MLNShapeSource(identifier: "marker-source", shape: point, options: nil)
            let shapeLayer = MLNSymbolStyleLayer(identifier: "marker-style", source: shapeSource)

            if let image = UIImage(named: "default-marker") {
                style.setImage(image, forName: "marker")
            }

            shapeLayer.iconImageName = NSExpression(forConstantValue: "marker")

            style.addSource(shapeSource)
            style.addLayer(shapeLayer)
        }
    }
}
```

Load the Radar style in your `Activity` and forward the lifecycle calls to the `MapView`:

```kotlin
import android.os.Bundle
import android.view.Gravity
import androidx.appcompat.app.AppCompatActivity
import androidx.core.content.res.ResourcesCompat
import androidx.core.graphics.drawable.toBitmap
import org.maplibre.android.MapLibre
import org.maplibre.android.camera.CameraPosition
import org.maplibre.android.geometry.LatLng
import org.maplibre.android.maps.MapView
// Optional imports for adding a marker.
import org.maplibre.android.plugins.annotation.SymbolManager
import org.maplibre.android.plugins.annotation.SymbolOptions

const val MARKER_NAME = "radar-marker"

class MainActivity : AppCompatActivity() {

    private lateinit var mapView: MapView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        val style = "radar-default-v1"
        val publishableKey = "prj_live_pk_..."
        val styleURL = "https://api.radar.io/maps/styles/$style?publishableKey=$publishableKey"

        // Must be called before the layout that contains the MapView is inflated.
        MapLibre.getInstance(this)

        setContentView(R.layout.activity_main)

        mapView = findViewById(R.id.mapView)
        mapView.onCreate(savedInstanceState)
        mapView.getMapAsync { map ->
            // Hide the MapLibre logo (the layout shows the Radar logo) and keep attribution visible.
            map.uiSettings.isLogoEnabled = false
            map.uiSettings.attributionGravity = Gravity.END or Gravity.BOTTOM
            map.uiSettings.setAttributionMargins(0, 0, 24, 24)

            map.setStyle(styleURL) { loadedStyle ->
                // Optionally add a marker once the style has loaded.
                val markerDrawable = ResourcesCompat.getDrawable(
                    resources,
                    R.drawable.default_marker,
                    null
                )!!
                loadedStyle.addImage(MARKER_NAME, markerDrawable.toBitmap())

                val symbolManager = SymbolManager(mapView, map, loadedStyle)
                // Draw the marker over map labels and icons.
                symbolManager.iconAllowOverlap = true
                symbolManager.iconIgnorePlacement = true

                val symbol = symbolManager.create(
                    SymbolOptions()
                        .withLatLng(LatLng(40.7342, -73.9911))
                        .withIconImage(MARKER_NAME)
                        .withIconSize(1.25f)
                        .withIconAnchor("bottom")
                )

                map.cameraPosition = CameraPosition.Builder()
                    .target(symbol.latLng)
                    .zoom(11.0)
                    .build()
            }
        }
    }

    override fun onStart() {
```

### React Native

The Radar React Native SDK exports a `Map` component that loads a Radar style, shows the Radar logo and map attribution, and by default shows the user's location when it is available. It depends on [MapLibre React Native](https://github.com/maplibre/maplibre-react-native). React Native SDK 4.39.0 requires `@maplibre/maplibre-react-native` 11.

Install both packages:

```bash
npm install react-native-radar @maplibre/maplibre-react-native
```

Complete the platform setup in the MapLibre React Native documentation, then initialize Radar and render the map inside a view that has a size:

```jsx
import { View } from 'react-native';
import Radar, { Map } from 'react-native-radar';

Radar.initialize('prj_live_pk_...');

export const App = () => (
  <View style={{ width: '100%', height: '100%' }}>
    <Map
      mapOptions={{
        mapStyle: 'radar-default-v1',
        showUserLocation: false,
        onDidFinishLoadingMap: () => console.log('map loaded'),
        onDidFailLoadingMap: () => console.log('map failed to load'),
        onRegionDidChange: (event) => console.log(event.center, event.zoom),
      }}
    />
  </View>
);
```

`mapOptions` accepts `mapStyle`, `showUserLocation` (default `true`), `mapRef`, `onRegionDidChange`, `onDidFinishLoadingMap`, `onWillStartLoadingMap`, and `onDidFailLoadingMap`. Children of `Map` are rendered inside the MapLibre map, so add markers and layers with MapLibre React Native components. For example, set the starting view with MapLibre's `Camera` component and add a marker with its `Marker` component, as the React Native tab in [Complete example](#complete-example) does. If MapLibre React Native isn't installed, the component logs a warning and renders nothing. See [React Native SDK](/sdk/react-native).

### Attribution

Radar's terms of use require the Radar logo on the map, and the map data requires attribution.

- **Radar web SDK**: The logo (bottom left) and the attribution control (bottom right) are added for you. Don't hide them with CSS or cover them with your own interface. When the map container is narrower than 380 pixels, the SDK hides the attribution text and keeps the logo.
- **React Native `Map` component**: The logo and attribution are added for you.
- **MapLibre GL JS, Leaflet, MapLibre Native, and Flutter**: Keep the library's attribution control enabled, and add the Radar logo yourself as the iOS and Android examples do. Logo files are in the [Radar map assets download](https://static.radar.com/downloads/radar-map-assets-v1.zip).
- **Static maps**: Don't crop or cover the Radar logo rendered into the image.

## Handle results and errors

A map has no response body. Your app reacts to its events and state.

| Signal | Meaning | What to do |
| --- | --- | --- |
| `load` event | The style and first tiles have loaded. Fires once. | Hide your loading state. Add markers and fit the view. |
| `map.loaded()` returns `true` | The map is fully loaded and idle. | Safe to read map state. |
| `style.load` event | A style finished loading, including after `map.setStyle()`. | Reapply projection settings. Lines and polygons you added belong to the previous style, so after a style change call `map.clearFeatures()` and add them again. |
| `error` event | A style, tile, font, or image request failed, or MapLibre hit a rendering error. The event's `error` property has the details. | See [Loading, errors, and recovery](#loading-errors-and-recovery). |
| `moveend` event | The user or your code finished moving the map. | Read `map.getCenter()` or `map.getBounds()` to load data for the new view. |
| Marker `click` event | The user selected a marker. | Open details for that location. The marker's popup, if any, opens on its own. |
| Feature `click` event | The user selected a line or polygon. | Read `feature.properties` for your data. |

Markers, features, and the map itself use `[longitude, latitude]` order. Radar REST responses and most `Radar.*` method parameters use separate `latitude` and `longitude` fields. Mixing the two orders is the most common cause of markers that land in the wrong place.

### Loading, errors, and recovery

| Situation | What you see | What to do |
| --- | --- | --- |
| The maps plugin isn't registered or its script didn't load | `Radar.ui` is undefined and `Radar.ui.map` throws a `TypeError`. | Register the plugin before you create a map, or load the plugin script after the core SDK script. |
| `Radar.initialize()` is called with no key or a secret key | `Radar.initialize()` throws a `RadarPublishableKeyError`. | Pass a publishable key. |
| The map is created before `Radar.initialize()` | A console warning that the publishable key isn't set, then failed style requests. | Initialize first. |
| Wrong or revoked key | The `error` event fires and no map appears. | Show a message in place of the map. Check the key and its environment. |
| Container has no height | An empty area. The SDK logs a warning when the container has no inline width or height. | Give the container an explicit size. If you size it with a stylesheet, the map renders correctly and you can ignore the warning. |
| Stylesheet missing | Controls, markers, and popups are unstyled or mispositioned. | Import `radar-maps.css`. |
| Container resized or shown after being hidden | The map is clipped or stretched. | Call `map.resize()` after the container changes size, for example when a tab or drawer opens. |
| Network drops after load | Already loaded tiles stay visible. New tiles fail and fire `error` events. | Don't replace a working map with an error message for a tile error. Tiles load again when the user moves the map after the connection returns. |
| Marker image fails to load | The default marker appears and the SDK logs an error. | Check the image URL and its cross-origin headers. |
| Duplicate feature ID | `addLine`, `addPolygon`, or `addPolyline` throws an error. | Use unique IDs, or call `feature.remove()` or `map.clearFeatures()` first. |
| Single-page app navigation | Map instances accumulate and the browser can run out of WebGL contexts. | Call `map.remove()` when the view unmounts. |

Because the `error` event also fires for a single failed tile, treat an error as fatal only when it arrives before `load`, as the [Create a map](#create-a-map) examples do. To retry after a fatal error, call `map.remove()` and create the map again.

On React Native, use `onDidFailLoadingMap`. On iOS, implement the MapLibre Native delegate method `mapViewDidFailLoadingMap(_:withError:)`. On Android, call `mapView.addOnDidFailLoadingMapListener()`. The native tabs in [Complete example](#complete-example) use all three.

## Test the integration

1. **Happy path**: Load the page with a Test publishable key. The loading message disappears, the map renders, markers appear at the right places, and the view fits them. Select a marker and confirm its popup.
2. **Wrong key**: Use `prj_test_pk_invalid`. Confirm your error message appears and the page has no blank map area.
3. **Coordinate order**: Add one location whose position you know. If it appears in the wrong hemisphere or in the ocean, latitude and longitude are swapped.
4. **No data**: Render the map with an empty locations list. The map still loads at your default center and `fitToMarkers()` does nothing.
5. **Narrow screen**: Use your browser's device toolbar at 360 pixels wide. The map fills its container, the logo stays visible, and popups fit on screen.
6. **Resize and visibility**: If the map lives in a tab, modal, or collapsible panel, open and close it and confirm the map fills the container.
7. **Navigation**: In a single-page app, visit and leave the map view several times and confirm no console errors about WebGL contexts.
8. **Offline**: Turn off the network in developer tools after the map loads, pan, then turn it back on and pan again. The map recovers without a reload.
9. **Live key**: Repeat the happy path with your Live publishable key before launch.

## Production checklist

- [ ] The page uses a Live publishable key supplied by configuration, not a secret key and not a hard-coded Test key.
- [ ] The SDK and plugin versions are pinned (CDN URLs include versions; npm installs use your lockfile).
- [ ] The container has an explicit size at every breakpoint, and `map.resize()` is called when it changes.
- [ ] A loading state and an error state exist, and the error state has a text fallback such as the address.
- [ ] The Radar logo and map attribution are visible.
- [ ] Maps are removed on unmount.
- [ ] Popup content built from your data uses `text` or DOM elements, not unescaped `html`.
- [ ] Custom style IDs and custom marker IDs exist in the Live environment's project.
- [ ] You have checked [Coverage](#coverage) for the countries you serve.
