## Complete example

Each tab renders a Radar map centered on Union Square in New York, adds one marker, and shows a message if the map fails to load.

```html JavaScript
<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <link href="https://js.radar.com/maps/v1.2.0/radar-maps.css" rel="stylesheet">
    <script src="https://js.radar.com/v5.3.0/radar.min.js"></script>
    <script src="https://js.radar.com/maps/v1.2.0/radar-maps.min.js"></script>
  </head>

  <body>
    <p id="map-status" role="status">Loading map...</p>
    <div id="map" style="width: 100%; height: 500px;"></div>

    <script>
      const statusEl = document.getElementById('map-status');
      let loaded = false;

      try {
        Radar.initialize('prj_test_pk_...');

        const map = Radar.ui.map({
          container: 'map',
          style: 'radar-default-v1',
          center: [-73.9905, 40.7359], // [longitude, latitude]
          zoom: 14,
        });

        map.on('load', () => {
          loaded = true;
          statusEl.hidden = true;

          Radar.ui.marker({ popup: { text: 'Union Square' } })
            .setLngLat([-73.9905, 40.7359])
            .addTo(map);
        });

        map.on('error', (event) => {
          console.error('Map error', event.error);
          if (!loaded) {
            statusEl.textContent = 'The map could not load. Refresh the page to try again.';
          }
        });
      } catch (err) {
        // Thrown when the key is missing or is a secret key, or when the plugin did not load.
        console.error(err);
        statusEl.textContent = 'The map could not load.';
      }
    </script>
  </body>
</html>
```

```jsx React
import { useEffect, useRef, useState } from 'react';
import Radar from 'radar-sdk-js';
import { createMapsPlugin } from '@radarlabs/plugin-maps';
import '@radarlabs/plugin-maps/dist/radar-maps.css';

Radar.registerPlugin(createMapsPlugin());
Radar.initialize(process.env.REACT_APP_RADAR_PUBLISHABLE_KEY);

const CENTER = [-73.9905, 40.7359]; // [longitude, latitude]

export default function RadarMap() {
  const containerRef = useRef(null);
  const [status, setStatus] = useState('loading');

  useEffect(() => {
    let loaded = false;

    const map = Radar.ui.map({
      container: containerRef.current,
      style: 'radar-default-v1',
      center: CENTER,
      zoom: 14,
    });

    map.on('load', () => {
      loaded = true;
      setStatus('ready');

      Radar.ui.marker({ popup: { text: 'Union Square' } })
        .setLngLat(CENTER)
        .addTo(map);
    });

    map.on('error', (event) => {
      console.error('Map error', event.error);
      if (!loaded) {
        setStatus('error');
      }
    });

    // Remove the map when the component unmounts.
    return () => map.remove();
  }, []);

  return (
    <div>
      {status === 'loading' && <p role="status">Loading map...</p>}
      {status === 'error' && <p role="alert">The map could not load. Refresh the page to try again.</p>}
      <div ref={containerRef} style={{ width: '100%', height: '500px' }} />
    </div>
  );
}
```

```swift Swift
import SwiftUI
import MapLibre

struct RadarMap: UIViewRepresentable {
    let publishableKey = "prj_test_pk_..."
    let center = CLLocationCoordinate2D(latitude: 40.7359, longitude: -73.9905)
    var onLoadFailed: (Error) -> Void

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    func makeUIView(context: Context) -> MLNMapView {
        let styleURL = URL(string: "https://api.radar.io/maps/styles/radar-default-v1?publishableKey=\(publishableKey)")

        let mapView = MLNMapView(frame: .zero, styleURL: styleURL)
        mapView.autoresizingMask = [.flexibleWidth, .flexibleHeight]
        mapView.setCenter(center, zoomLevel: 14, animated: false)
        mapView.delegate = context.coordinator

        // The Radar logo replaces the MapLibre logo. Add radar-logo to your asset catalog.
        mapView.logoView.isHidden = true
        let logoImageView = UIImageView(image: UIImage(named: "radar-logo"))
        logoImageView.translatesAutoresizingMaskIntoConstraints = false
        mapView.addSubview(logoImageView)
        NSLayoutConstraint.activate([
            logoImageView.bottomAnchor.constraint(equalTo: mapView.safeAreaLayoutGuide.bottomAnchor, constant: -10),
            logoImageView.leadingAnchor.constraint(equalTo: mapView.safeAreaLayoutGuide.leadingAnchor, constant: 10),
            logoImageView.widthAnchor.constraint(equalToConstant: 74),
            logoImageView.heightAnchor.constraint(equalToConstant: 26)
        ])

        return mapView
    }

    func updateUIView(_ uiView: MLNMapView, context: Context) {
    }

    class Coordinator: NSObject, MLNMapViewDelegate {
        let control: RadarMap

        init(_ control: RadarMap) {
            self.control = control
        }

        // Add the marker when the style finishes loading. Add default-marker to your asset catalog.
        func mapView(_ mapView: MLNMapView, didFinishLoading style: MLNStyle) {
            let point = MLNPointAnnotation()
            point.coordinate = control.center

            let shapeSource = MLNShapeSource(identifier: "marker-source", shape: point, options: nil)
            let shapeLayer = MLNSymbolStyleLayer(identifier: "marker-style", source: shapeSource)

            if let image = UIImage(named: "default-marker") {
                style.setImage(image, forName: "marker")
            }
            shapeLayer.iconImageName = NSExpression(forConstantValue: "marker")

            style.addSource(shapeSource)
            style.addLayer(shapeLayer)
        }

        func mapViewDidFailLoadingMap(_ mapView: MLNMapView, withError error: Error) {
            control.onLoadFailed(error)
        }
    }
}
```

```kotlin Kotlin
import org.maplibre.android.plugins.annotation.SymbolOptions

class MapActivity : AppCompatActivity() {

    private lateinit var mapView: MapView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        val publishableKey = "prj_test_pk_..."
        val styleURL = "https://api.radar.io/maps/styles/radar-default-v1?publishableKey=$publishableKey"
        val center = LatLng(40.7359, -73.9905)

        // Must be called before the layout is inflated. The layout holds the MapView and the Radar logo.
        MapLibre.getInstance(this)
        setContentView(R.layout.activity_main)

        mapView = findViewById(R.id.mapView)
        mapView.onCreate(savedInstanceState)

        mapView.addOnDidFailLoadingMapListener {
            Toast.makeText(this, "The map could not load.", Toast.LENGTH_LONG).show()
        }

        mapView.getMapAsync { map ->
            map.uiSettings.isLogoEnabled = false
            map.cameraPosition = CameraPosition.Builder().target(center).zoom(14.0).build()

                loadedStyle.addImage("radar-marker", markerDrawable.toBitmap())

                val symbolManager = SymbolManager(mapView, map, loadedStyle)
                symbolManager.iconAllowOverlap = true
                symbolManager.iconIgnorePlacement = true
                symbolManager.create(
                    SymbolOptions()
                        .withLatLng(center)
                        .withIconImage("radar-marker")
                        .withIconAnchor("bottom")
                )
            }
        }
    }

    override fun onResume() { super.onResume(); mapView.onResume() }

    override fun onStop() { super.onStop(); mapView.onStop() }
    override fun onLowMemory() { super.onLowMemory(); mapView.onLowMemory() }
    override fun onDestroy() { super.onDestroy(); mapView.onDestroy() }

        super.onSaveInstanceState(outState)
```

## Prerequisites

- A publishable key. See [Choose the right API key](/fundamentals/authentication-and-access#choose-the-right-api-key). Never put a secret key in client code.
- **Web**: A browser with WebGL. The examples use `radar-sdk-js` 5.3.0 and `@radarlabs/plugin-maps` 1.2.0. The plugin requires `radar-sdk-js` 5.1.0 or later.
- **Native**: MapLibre Native or the Radar React Native SDK. See [Compatibility](#compatibility).

## Load the SDK and stylesheet

Maps are a plugin to the web SDK. With script tags, load the stylesheet, the core SDK, and the plugin, in that order, as the JavaScript tab does. The plugin script registers itself and bundles MapLibre GL JS.

With npm, install the core SDK, the plugin, and MapLibre GL JS:

```bash
npm install radar-sdk-js @radarlabs/plugin-maps maplibre-gl
```

Then import the plugin stylesheet and register the plugin before you use `Radar.ui`, as the React tab does. Without the stylesheet, controls, markers, and popups are mispositioned.

## Size the container

Give the map container an explicit width and height. A container with no height renders nothing. If the container changes size later, call `map.resize()`.

## Create a map

Call `Radar.initialize()` with your publishable key, then pass the container's ID or element to `Radar.ui.map()`. Coordinates are `[longitude, latitude]`. Swapping them is the most common cause of a marker in the wrong place.

| Option | Default | Description |
| --- | --- | --- |
| `container` | None | Required. The element ID or element to render into. |
| `style` | `radar-default-v1` | A [style](#styles) name, a custom style ID, a style URL, or a MapLibre style object. |
| `center` | Estimated from the client's IP address | The initial center. The IP estimate is approximate. |
| `zoom` | Set by the style | The initial zoom level. Higher numbers are closer. |
| `language` | None | Label language: a code such as `en`, or `local` for the browser's language. |
| `showZoomControls` | `true` | Whether to show the zoom buttons. |

`Radar.ui.map()` returns a MapLibre GL JS [`Map`](https://maplibre.org/maplibre-gl-js/docs/API/classes/Map/), so it also accepts any MapLibre [MapOptions](https://maplibre.org/maplibre-gl-js/docs/API/type-aliases/MapOptions/) except `transformRequest`, and every MapLibre method and event works on it. In a single-page app, call `map.remove()` when the view unmounts.

## Add a marker

The example adds its marker in the `load` handler. `Radar.ui.marker()` returns a MapLibre [`Marker`](https://maplibre.org/maplibre-gl-js/docs/API/classes/Marker/). Position it with `setLngLat()` and add it with `addTo(map)`.

| Option | Description |
| --- | --- |
| `color`, `scale` | Color (default `#000257`) and scale (default `1`) of the default marker. |
| `url`, `marker` | An image URL, or the ID of a marker image uploaded to a [custom style](#custom-styles), with optional `width` and `height`. Without a size, the image is capped at 64 by 64 pixels. |
| `element` | A DOM element to use as the marker. |
| `popup` | Options for a [popup](#add-popups) that opens when the marker is selected. |

The plugin forwards only `color`, `element`, and `scale` from MapLibre's marker options. For other MapLibre marker behavior, call the marker's methods, for example `marker.setDraggable(true)`.

## Handle load failure

Show a loading message until `load` fires, and listen for the map's `error` event. That event also fires for a single failed tile, so treat an error as fatal only when it arrives before `load`, as the example does. To retry, call `map.remove()` and create the map again. The native tabs use `mapViewDidFailLoadingMap(_:withError:)` on iOS, `addOnDidFailLoadingMapListener()` on Android, and `onDidFailLoadingMap` on React Native.

## Fit the view to markers

`map.fitToMarkers()` centers and zooms the map so every marker is visible. It accepts MapLibre fit options such as `padding` and `maxZoom`.

```javascript
locations.forEach((location) => {
  Radar.ui.marker({ popup: { text: location.name } })
    .setLngLat(location.coordinates) // [longitude, latitude]
    .addTo(map);
});

map.fitToMarkers({ maxZoom: 15, padding: 60 });
```

`map.getMarkers()` returns the markers on the map and `map.clearMarkers()` removes them.

## Add popups

Attach a popup to a marker with the marker's `popup` option, or create a standalone one with `Radar.ui.popup()`. Set its content with `text`, `html`, or `element`. Popups also accept MapLibre popup options.

```javascript
Radar.ui.popup({ text: 'Flatiron' })
  .setLngLat([-73.9897, 40.7411])
  .addTo(map);
```

<Warning>
  `html` is inserted as markup. If any part of the content comes from user input or data you don't control, use `text`, or an `element` built with `textContent`.
</Warning>

## Styles

Pass a style name as the `style` option, or call `map.setStyle()` to switch later.

| Style | Description |
| --- | --- |
| `radar-default-v1` | The default, with street-level detail. |
| `radar-light-v1` | Light monochrome, for highlighting your own data. |
| `radar-dark-v1` | Dark monochrome. |
| `radar-satellite-v1` | Satellite imagery. Available on the [Enterprise plan](https://radar.com/pricing). |
| `radar-hybrid-v1` | Satellite imagery with roads and labels. Available on the Enterprise plan. |

### Custom styles

To match the map to your brand, open **Maps** > **Styles** in the dashboard, click **Create style**, choose a base style, and edit its layers, colors, fonts, and marker images. When you click **Save**, the confirmation dialog shows the style ID. Pass that ID as the `style` option:

```javascript
const map = Radar.ui.map({
  container: 'map',
  style: 'CUSTOM_STYLE_ID',
});
```

Changes to a style that is already in use are visible to users as soon as you save.

## Display a polygon

Pass a GeoJSON `Polygon` or `MultiPolygon` feature to `map.addPolygon()`. Each feature needs a unique `id`, and adding a second feature with the same ID throws an error.

```javascript
const feature = map.addPolygon({
  type: 'Feature',
  id: 'central-park',
  properties: { name: 'Central Park' },
  geometry: {
    type: 'Polygon',
    coordinates: [[
      [-73.9582, 40.8003],
      [-73.9817, 40.7682],
      [-73.9728, 40.7642],
      [-73.9492, 40.7968],
      [-73.9582, 40.8003],
    ]],
  },
}, {
  paint: { 'fill-color': 'blue', 'fill-opacity': 0.4 },
});

feature.on('click', ({ feature, originalEvent }) => {
  Radar.ui.popup({ text: feature.properties.name })
    .setLngLat(originalEvent.lngLat)
    .addTo(map);
});

map.fitToFeatures({ padding: 40 });
```

`paint` takes `fill-color`, `fill-opacity`, `border-width`, `border-color`, and `border-opacity`. Remove a feature with `feature.remove()`, or all of them with `map.clearFeatures()`.

## Display a route

Add a GeoJSON `LineString` with `map.addLine()`, or an encoded polyline with `map.addPolyline()`. With `geometry: 'linestring'`, each route from `Radar.distance()` has a `geometry` you can pass to `addLine()`:

```javascript
const { routes } = await Radar.distance({
  origin,       // { latitude, longitude }
  destination,  // { latitude, longitude }
  modes: ['car'],
  geometry: 'linestring',
});

if (routes.car && routes.car.geometry) {
  map.addLine({ type: 'Feature', id: 'route', geometry: routes.car.geometry }, {
    paint: { 'line-color': '#000257', 'line-width': 5 },
  });
  map.fitToFeatures({ padding: 40 });
}
```

## Static maps

A static map is an image that Radar renders from a URL, for emails, receipts, and other places where you can't run a map library. The key goes in the `publishableKey` query parameter.

```text
https://api.radar.io/maps/static?width=600&height=400&center=40.73430,-73.99110&zoom=15&style=radar-default-v1&scale=2&markers=color:0x000257|40.73430,-73.99110&publishableKey=prj_live_pk_...
```

This returns a 600 by 400 image at double resolution with one marker. For every parameter, including marker and path styling, see [Static maps](/api/maps#static-maps).

## Compatibility

Radar maps use MapLibre styles, so any library that supports them can display a Radar map. Load the style by URL, with your publishable key in the query string:

```text
https://api.radar.io/maps/styles/radar-default-v1?publishableKey=prj_live_pk_...
```

Outside the Radar web SDK and the React Native `Map` component, keep the library's attribution control visible and add the Radar logo to the map yourself. Logo and marker images are in the [Radar map assets download](https://static.radar.com/downloads/radar-map-assets-v1.zip). See [Attribution](/api/maps#attribution).

### MapLibre GL JS

Set the Radar style URL as the map's `style`:

```javascript
const map = new maplibregl.Map({
  container: 'map',
  style: 'https://api.radar.io/maps/styles/radar-default-v1?publishableKey=prj_live_pk_...',
  center: [-122.65, 45.52],
  zoom: 9,
});
```

### Leaflet

Use the [maplibre-gl-leaflet](https://github.com/maplibre/maplibre-gl-leaflet) plugin with the same style URL:

```javascript
L.maplibreGL({
```

## Coverage

Radar map data comes from [OpenStreetMap](https://www.openstreetmap.org). Maps coverage is rated great for consumer and logistics use cases in the US, Canada, and the UK. In other countries it is rated great for consumer use cases, such as store locators, and good for logistics use cases, such as delivery.

## Handle results and errors

| Signal | What it means | What to do |
| --- | --- | --- |
| `load` event | The style and first tiles have loaded. | Hide the loading state, add markers, and fit the view. |
| `error` event before `load` | The map couldn't load, for example with a wrong or revoked key. | Show a message in place of the map. Check the key and its environment. |
| `error` event after `load` | A tile, font, or image request failed. | Keep the map. Tiles load again when the user moves the map after the connection returns. |
| `Radar.initialize()` throws | The key is missing or is a secret key. | Pass a publishable key. |
| `Radar.ui` is undefined | The maps plugin isn't registered, or its script didn't load. | Register the plugin, or load its script after the core SDK. |
| Empty area, no error | The container has no height. | Give the container an explicit size. |

## Test the integration

- **Passing case**: Load the page with a Test publishable key. The loading message disappears, the map renders, and selecting the marker opens its popup.
- **Failing case**: Replace the key with `prj_test_pk_invalid` and reload. Your error message appears in place of a blank area.

For more, see [Test, troubleshoot, and go live](/fundamentals/testing-and-launch).

## Production checklist

- [ ] The page uses a Live publishable key from configuration, not a secret key or a hard-coded Test key.
- [ ] SDK and plugin versions are pinned.
- [ ] The container has an explicit size at every breakpoint, and maps are removed on unmount.
- [ ] A loading state and an error state exist.
- [ ] The Radar logo and map attribution are visible.
- [ ] Popup content built from your data uses `text` or DOM elements, not unescaped `html`.
