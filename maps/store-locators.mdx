By the end of this guide you have a location finder in a web page. A visitor types an address (no location permission needed) or chooses to share their position, sees the nearest locations in a list and on a map, selects one, and sees its details with the driving distance and time.

```html JavaScript
<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>Find a location</title>
    <link href="https://js.radar.com/maps/v1.2.0/radar-maps.css" rel="stylesheet">
    <link href="https://js.radar.com/autocomplete/v1.2.0/radar-autocomplete.css" rel="stylesheet">
    <script src="https://js.radar.com/v5.3.0/radar.min.js"></script>
    <script src="https://js.radar.com/maps/v1.2.0/radar-maps.min.js"></script>
    <script src="https://js.radar.com/autocomplete/v1.2.0/radar-autocomplete.min.js"></script>
    <style>
      body { margin: 0; font-family: system-ui, sans-serif; }
      .locator { display: flex; height: 100vh; }
      .panel { width: 360px; padding: 16px; box-sizing: border-box; overflow-y: auto; }
      .map-wrap { flex: 1; }
      #results { list-style: none; margin: 0; padding: 0; }
      #results button {
        display: block; width: 100%; padding: 12px 0; border: 0;
        border-bottom: 1px solid #ddd; background: none;
        text-align: left; font: inherit; cursor: pointer;
      }
      #results button[aria-current="true"] { font-weight: 600; }
      .distance, .detail { color: #555; }

      /* Stack the list above the map on narrow screens. */
      @media (max-width: 700px) {
        .locator { flex-direction: column; }
        .panel { width: 100%; height: 50vh; }
        .map-wrap { flex: none; height: 50vh; }
      }
    </style>
  </head>

  <body>
    <div class="locator">
      <div class="panel">
        <div id="autocomplete"></div>
        <p><button id="use-location" type="button">Use my location</button></p>
        <p id="status" role="status" aria-live="polite">Search by address, city, or postal code.</p>
        <ul id="results"></ul>
      </div>
      <div class="map-wrap">
        <div id="map" style="width: 100%; height: 100%;"></div>
      </div>
    </div>

    <script>
      const TAG = 'store';
      const RADIUS_METERS = 10000;
      const LIMIT = 10;

      const statusEl = document.getElementById('status');
      const resultsEl = document.getElementById('results');
      let searchCount = 0;

      Radar.initialize('prj_test_pk_...');

      // With no center, the map starts at an estimate of the visitor's area from their IP address.
      const map = Radar.ui.map({ container: 'map', style: 'radar-default-v1' });

      function setStatus(message) {
        statusEl.textContent = message;
      }

      // Straight-line distance in miles between two { latitude, longitude } points.
      function straightLineMiles(a, b) {
        const toRadians = (degrees) => (degrees * Math.PI) / 180;
        const dLat = toRadians(b.latitude - a.latitude);
        const dLng = toRadians(b.longitude - a.longitude);
        const h = Math.sin(dLat / 2) ** 2
          + Math.cos(toRadians(a.latitude)) * Math.cos(toRadians(b.latitude)) * Math.sin(dLng / 2) ** 2;
        return 3958.8 * 2 * Math.asin(Math.sqrt(h));
      }

      // Convert a Radar geofence into the shape the list and the map both use.
      function toLocation(geofence, origin) {
        const [longitude, latitude] = geofence.geometryCenter.coordinates;
        const location = {
          id: geofence.externalId,
          name: geofence.description,
          metadata: geofence.metadata || {},
          latitude,
          longitude,
        };
        location.miles = straightLineMiles(origin, location);
        return location;
      }

      function renderResult(location) {
        const item = document.createElement('li');
        const button = document.createElement('button');
        button.type = 'button';

        const name = document.createElement('div');
        name.textContent = location.name;

        const distance = document.createElement('div');
        distance.className = 'distance';
        distance.textContent = `${location.miles.toFixed(1)} mi away (straight line)`;

        button.append(name, distance);

        ['hours', 'phone'].forEach((key) => {
          if (location.metadata[key]) {
            const detail = document.createElement('div');
            detail.className = 'detail';
            detail.textContent = String(location.metadata[key]);
            button.append(detail);
          }
        });

        item.append(button);
        resultsEl.append(item);
        return button;
      }

      async function selectLocation(origin, location, marker, button, fromMarker) {
        resultsEl.querySelectorAll('button').forEach((el) => el.removeAttribute('aria-current'));
        button.setAttribute('aria-current', 'true');
        button.scrollIntoView({ block: 'nearest' });

        // A marker click already opened its own popup. A list click has to open it.
        if (!fromMarker) {
          map.getMarkers().forEach((other) => {
            if (other !== marker && other.getPopup().isOpen()) {
              other.togglePopup();
            }
          });
          if (!marker.getPopup().isOpen()) {
            marker.togglePopup();
          }
        }

        map.flyTo({ center: [location.longitude, location.latitude], zoom: 15 });

        // Route distance and travel time for the selected location only.
        try {
          const { routes } = await Radar.distance({
            origin,
            destination: { latitude: location.latitude, longitude: location.longitude },
            modes: ['car'],
            units: 'imperial',
          });
          if (routes.car && routes.car.distance && routes.car.duration) {
            button.querySelector('.distance').textContent =
              `${routes.car.distance.text} by car, about ${routes.car.duration.text}`;
          }
        } catch (err) {
          // Keep the straight-line distance if the route request fails.
          console.error(err);
        }
      }

      async function searchNear(origin, label) {
        const thisSearch = ++searchCount;
        setStatus('Searching...');
        resultsEl.replaceChildren();
        map.clearMarkers();

        let geofences;
        try {
          ({ geofences } = await Radar.searchGeofences({
            near: origin,
            radius: RADIUS_METERS,
            tags: [TAG],
            limit: LIMIT,
            includeGeometry: false,
          }));
        } catch (err) {
          if (thisSearch !== searchCount) return;
          console.error(err);
          setStatus(err.status === 'ERROR_RATE_LIMIT'
            ? 'Too many searches. Wait a moment and try again.'
            : 'Search is unavailable right now. Try again.');
          return;
        }

        if (thisSearch !== searchCount) return; // a newer search has started

        if (geofences.length === 0) {
          setStatus(`No locations found near ${label}. Try another address.`);
          map.flyTo({ center: [origin.longitude, origin.latitude], zoom: 10 });
          return;
        }

        // One array drives both the list and the map, so they cannot disagree.
        const locations = geofences.map((geofence) => toLocation(geofence, origin));

        locations.forEach((location) => {
          const marker = Radar.ui.marker({ popup: { text: location.name } })
            .setLngLat([location.longitude, location.latitude])
            .addTo(map);
          const button = renderResult(location);

          button.addEventListener('click', () => selectLocation(origin, location, marker, button, false));
          marker.on('click', () => selectLocation(origin, location, marker, button, true));
        });

        map.fitToMarkers({ maxZoom: 14, padding: 60 });
        setStatus(`${locations.length} locations near ${label}, nearest first.`);
      }

      // Search by address. Works with no location permission.
      Radar.ui.autocomplete({
        container: 'autocomplete',
        width: '100%',
        placeholder: 'Address, city, or postal code',
        onSelection: (address) => {
          searchNear(
            { latitude: address.latitude, longitude: address.longitude },
            address.formattedAddress,
          );
        },
        onError: (err) => {
          console.error(err);
          setStatus('Address search is unavailable right now. Try again.');
        },
      });

      // Search by the visitor's position, only when they ask for it.
      document.getElementById('use-location').addEventListener('click', async () => {
        setStatus('Getting your location...');
        try {
          const { latitude, longitude } = await Radar.getLocation();
          searchNear({ latitude, longitude }, 'your location');
        } catch (err) {
          setStatus(err.status === 'ERROR_PERMISSIONS'
            ? 'Location permission is off. Search by address instead.'
            : 'Your location is unavailable. Search by address instead.');
        }
      });

      // Optional: start with results near an estimate of the visitor's area.
      Radar.ipGeocode()
        .then(({ address }) => {
          if (address && searchCount === 0) {
            searchNear(
              { latitude: address.latitude, longitude: address.longitude },
              address.city ? `${address.city} (estimated from your network)` : 'your approximate area',
            );
          }
        })
        .catch(() => {
          // No estimate available. The visitor can still search by address.
        });
    </script>
  </body>
</html>
```

```swift Swift
import CoreLocation
import RadarSDK
import UIKit

@main
class AppDelegate: UIResponder, UIApplicationDelegate {
    func application(
        _ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?
    ) -> Bool {
        Radar.initialize(publishableKey: "prj_test_pk_...")
        return true
    }
}

func findStores(near query: String) {
    // Search by address. Works with no location permission.
    Radar.geocode(address: query) { (status, addresses) in
        guard status == .success, let address = addresses?.first else {
            // Show that the address could not be found.
            return
        }

        let origin = CLLocation(latitude: address.coordinate.latitude, longitude: address.coordinate.longitude)

        Radar.searchGeofences(
            near: origin,
            radius: 10000,
            tags: ["store"],
            metadata: nil,
            limit: 10,
            includeGeometry: false
        ) { (status, _, geofences) in
            guard status == .success, let geofences = geofences else {
                // Show the error state.
                return
            }

            if geofences.isEmpty {
                // Show the empty state.
                return
            }

            // Geofences are sorted by distance from the origin. Render the list and the map from this one array.
            for geofence in geofences {
                guard let circle = geofence.geometry as? RadarCircleGeometry else { continue }
                let center = circle.center.coordinate
                let meters = origin.distance(from: CLLocation(latitude: center.latitude, longitude: center.longitude))
                print(geofence.externalId ?? "", geofence.__description, String(format: "%.1f mi away (straight line)", meters / 1609.344))
            }
        }
    }
}
```

```kotlin Kotlin
    Radar.geocode(query) { geocodeStatus, addresses ->
        val address = addresses?.firstOrNull()
        if (geocodeStatus != Radar.RadarStatus.SUCCESS || address == null) {

            return@geocode

            latitude = address.coordinate.latitude

            tags = arrayOf("store"),
            metadata = null,
            limit = 10,

            if (status != Radar.RadarStatus.SUCCESS || geofences == null) {
                // Show the error state.
                return@searchGeofences

                // Show the empty state.
                return@searchGeofences

            }
        }
```

```javascript React Native
import Radar from 'react-native-radar';

Radar.initialize('prj_test_pk_...');

export async function findStores(query) {
  try {
    // Search by address. Works with no location permission.
    const { addresses } = await Radar.geocode({ address: query });
    if (!addresses || addresses.length === 0) {
      // Show that the address could not be found.
      return [];
    }

    const origin = { latitude: addresses[0].latitude, longitude: addresses[0].longitude };

    const { geofences } = await Radar.searchGeofences({
      near: origin,
      radius: 10000,
      tags: ['store'],
      limit: 10,
      includeGeometry: false,
    });

    if (!geofences || geofences.length === 0) {
      // Show the empty state.
      return [];
    }

    // Geofences are sorted by distance from the origin. Render the list and the map from this one array.
    return geofences.map((geofence) => {
      const [longitude, latitude] = geofence.geometryCenter.coordinates;
      return {
        id: geofence.externalId,
        name: geofence.description,
        metadata: geofence.metadata || {},
        latitude,
        longitude,
      };
    });
  } catch (err) {
    // Show the error state. The promise rejects with a Radar status such as ERROR_NETWORK in err.code.
    console.error(err.code);
    return [];
  }
```

## Prerequisites

- A Radar account and a publishable key. See [Keys](#keys).
- Your locations, each with a name, coordinates or an address, and an ID from your own system.
- **Web**: The examples use `radar-sdk-js` 5.3.0, `@radarlabs/plugin-maps` 1.2.0, and `@radarlabs/plugin-autocomplete` 1.2.0. The autocomplete plugin requires `radar-sdk-js` 5.3.0 or later.
- **Permissions**: None. Address search works without location permission. The "Use my location" button asks for browser location permission only when the visitor selects it.
- **Access**: To import geofences in the Radar dashboard, you need access to the project's Geofences page. To import through the API, you need a secret key on your backend.

## How it works

| Part | Responsibility |
| --- | --- |
| Radar (setup) | Stores your locations as geofences. Each has a `tag` (such as `store`), an `externalId` (your ID), a `description` (the display name), and `metadata` (hours, phone, services). |
| Your page | Collects a search origin from the autocomplete component or the browser's location, calls the search, and renders the list and the map from the same result array. |
| Radar (search) | Returns the geofences near the origin, nearest first, filtered by tag. Separately returns route distance and travel time on request. |
| Your backend | Keeps Radar in step with your location database through import, the API, or nightly sync. It isn't in the request path of a search. |

Two different distances appear in a locator, and they aren't interchangeable:

| Question | Straight-line distance | Route distance and travel time |
| --- | --- | --- |
| What it is | The distance between two points over the earth's surface, ignoring roads. | The distance along roads for a travel mode, and the estimated time to travel it. |
| Where it comes from | Geofence search orders results by it. You compute the number in the browser. | `Radar.distance()` for one destination or `Radar.matrix()` for many. |
| Cost | No extra request. | One extra request per selection or per result set. |
| Label it as | "0.4 mi away" or "straight line". | "0.5 mi by car, about 3 mins". |

The nearest location in a straight line isn't always the quickest to reach, for example across a river or a highway. If you show both numbers, label each one.

## Build the locator

The locator is one page: an address input, a "Use my location" button, a results list, and a map, all driven by one search function.

To run the JavaScript tab in [Complete example](#complete-example), save it as an HTML file, replace the key, and open it in a browser. A finished search looks like this: the status line reads "3 locations near 841 Broadway, New York, NY 10003 USA, nearest first.", the list shows "Store 42, 0.4 mi away (straight line)" and two more entries, and the map shows three markers fitted to the view. Selecting "Store 42" highlights it, opens its popup on the map, and changes its distance line to "0.5 mi by car, about 3 mins".

For npm projects, install and register the two plugins as shown in [Display maps](/maps/maps#with-npm-or-yarn) and [Autocomplete, geocode, and validate addresses](/maps/addresses#add-the-autocomplete-component). The calls are the same.

### Accept an address

`Radar.ui.autocomplete()` renders a search input into the `autocomplete` container. As the visitor types, it suggests addresses, cities, and postal codes. When they choose a suggestion, `onSelection` receives an address object with `latitude`, `longitude`, and `formattedAddress`, and the example uses those coordinates as the search origin.

This is the primary path and it needs no location permission. The component waits for three characters before it searches, shows "No results" in its dropdown when nothing matches, and calls `onError` if a request fails. It fires `onSelection` only when the visitor picks a suggestion. If you build your own search box and want to accept free text on submit, call `Radar.forwardGeocode({ query })` and use the first address, or show a message when the `addresses` array is empty. For the component's options, including limiting results to a country, see [Autocomplete, geocode, and validate addresses](/maps/addresses).

### Accept the user's location

The "Use my location" button calls `Radar.getLocation()`, which triggers the browser's permission prompt the first time. Call it from a button, not on page load, so the visitor understands why they are being asked.

If the visitor denies permission, the promise rejects with an error whose `status` is `ERROR_PERMISSIONS`. If the browser can't determine a position or times out, the `status` is `ERROR_LOCATION`. In both cases the example tells the visitor to search by address and leaves the rest of the page working. See [Understand foreground and background location permissions](/fundamentals/location-permissions) for the browser and mobile permission models.

### Start near the visitor without asking for permission

The last block of the example calls `Radar.ipGeocode()` and, if it returns an address, runs a first search around it. An IP-derived location is an estimate of the visitor's city or region, taken from their network. It can be wrong by a large distance, especially on mobile networks, corporate networks, and VPNs. Use it to make the first screen useful, label it as an estimate, and let an address search or the visitor's own location replace it. Don't present it as the visitor's position.

### Search nearby locations

`Radar.searchGeofences()` returns the geofences nearest to `near`, sorted by straight-line distance.

| Parameter | Value in the example | Notes |
| --- | --- | --- |
| `near` | The search origin as `{ latitude, longitude }` | If you omit it, the SDK requests the browser's location, which triggers a permission prompt. Always pass it in a locator. |
| `radius` | `10000` | Meters. The maximum is `10000`. |
| `tags` | `['store']` | Only geofences with one of these tags are returned. |
| `limit` | `10` | The number of results to show. |
| `includeGeometry` | `false` | The locator needs only each geofence's center, which is always returned as `geometryCenter`. Leaving out the full shapes keeps responses small. |

Each returned geofence has `_id`, `description`, `tag`, `externalId`, `metadata`, and `geometryCenter`, a GeoJSON point in `[longitude, latitude]` order. In TypeScript, the SDK's `RadarGeofence` type declares only the first five, so extend the type or cast to read `geometryCenter`. For every parameter and field, see [Search geofences](/api/search#search-geofences).

To filter on metadata, such as locations that offer pickup, filter the returned array on `geofence.metadata` in the browser, or use a separate tag for those locations. The REST endpoint also accepts `metadata[key]=value` filters.

### Show the list and the map

The example converts the response into one `locations` array and renders both the list and the markers from it in a single loop. That is what keeps the two views consistent: the same locations, in the same order, with each list button holding a reference to its marker. Before each new search it clears the list and calls `map.clearMarkers()`, then calls `map.fitToMarkers()` so every result is visible.

List content is written with `textContent` and popups use the `text` option, so names and metadata from your data are never interpreted as markup.

A counter (`searchCount`) discards the response of an older search if the visitor starts a new one before it returns.

### Select a location

Selecting a list entry and selecting a marker both call `selectLocation()`, so they lead to the same state: the entry is marked with `aria-current`, scrolled into view, its popup is open, and the map moves to it. A marker click opens its own popup and closes the others, so the function only manages popups when the selection came from the list.

To open a full details page, add a link to each entry built from `location.id`, which is your own `externalId`, for example `/stores/store-42`. Use `metadata` for the fields you show inline, such as hours, phone number, and services.

### Show route distance and travel time

When a location is selected, the example calls `Radar.distance()` with the search origin and the location and replaces the straight-line label with the driving distance and time from `routes.car.distance.text` and `routes.car.duration.text`. If the request fails, or no drivable route exists, the straight-line label stays.

To show travel time for every result at once, make one `Radar.matrix()` request after the search, as in this example:

```javascript
const { matrix } = await Radar.matrix({
  origins: [origin],
  destinations: locations.map(({ latitude, longitude }) => ({ latitude, longitude })),
  mode: 'car',
  units: 'imperial',
});

// matrix[0][i] is the route from the origin to locations[i].
matrix[0].forEach((route, index) => {
  locations[index].drive = route.duration ? route.duration.text : null;
});
```

The search results are ordered by straight-line distance. If you re-sort the list by travel time, re-sort the single `locations` array before you render, so the list and the map stay in the same order. For units, travel modes, and request limits, see [Distance](/api/routing#distance) and [Matrix](/api/routing#matrix). For how the two distances relate, see [Straight-line distance and route distance](/api/routing#straight-line-distance-and-route-distance).

## Variations

### Branches, ATMs, and other location types

A bank finder, a dealer finder, and a pickup-point finder are the same flow with different tags and metadata.

- Import each type with its own tag, for example `branch` and `atm`.
- Search several types at once with `tags: ['branch', 'atm']`, or one type when the visitor picks a filter. Each result's `tag` tells you which type it is, so you can use a different marker color or icon per type with the marker's `color` or `url` option.
- Put type-specific attributes in metadata, for example `"driveThrough": true` or `"deposits": false`, and filter the result array on them.
- Geofences can carry operating hours. Search results include them as `operatingHours` when set.

### Use your own location list

If your locations already come from your own API, you can skip geofences. Keep the same structure: build one `locations` array, compute `straightLineMiles()` from the search origin to each, sort by it, take the first few, and render the list and markers from that array. Address search, the user's location, selection, and route distance work as in the example. This suits a small number of locations. With many locations, a geofence search avoids sending the whole list to every visitor.

### Search places

To find public places, such as a category of business or a named chain, use `Radar.searchPlaces()` with `chains` or `categories` in place of `Radar.searchGeofences()`. Results have a `name`, `categories`, and a `location` point. See [Search places](/api/search#search-places).

### Native apps

The same flow works in a native app: a search field, a nearby search, a list, and a map. The Radar iOS, Android, and React Native SDKs provide `Radar.searchGeofences`, `Radar.autocomplete`, `Radar.geocode`, and `Radar.getDistance`, and [Display maps](/maps/maps#compatibility) shows how to render a Radar map with MapLibre Native. The native tabs in [Complete example](#complete-example) geocode a typed address and search near it. These examples show the nearby search on its own:

```swift Swift
import RadarSDK

func searchStores(near origin: CLLocation) {
    Radar.searchGeofences(
        near: origin,
        radius: 10000,
        tags: ["store"],
        metadata: nil,
        limit: 10,
        includeGeometry: false

        guard status == .success, let geofences = geofences else {

            return

        if geofences.isEmpty {
            // Show the empty state.
            return
        }

        // Geofences are sorted by distance from the origin.

                let coordinate = circle.center.coordinate

            }
        }
    }
}
```

## Handle results and errors

`Radar.searchGeofences()` resolves with a `geofences` array. This is the raw API response for one result with `includeGeometry` set to `false`:

```json
{
  "meta": {
    "code": 200
  },
  "geofences": [
    {
      "_id": "64f0c0ffee0000000000002a",
      "live": false,
      "type": "circle",
      "tag": "store",
      "externalId": "store-42",
      "description": "Store 42",
      "metadata": {
        "phone": "555-0142",
        "hours": "8am to 9pm",
        "pickup": false
      },
      "geometryCenter": {
        "type": "Point",
        "coordinates": [-73.9897, 40.7411]
      },
      "geometryRadius": 100
    }
  ]
}
```

| Result | What your app does |
| --- | --- |
| One or more geofences | Render the list and markers from the array, in the order returned. |
| An empty `geofences` array | Show the empty state and keep the search box usable. This is a successful response, not an error. |
| `routes.car` present in a distance response | Show `distance.text` and `duration.text`, labeled as driving. |
| `routes.car` missing | No route was found for that mode. Keep the straight-line distance. |
| `address` missing from `Radar.ipGeocode()` | No estimate is available. Wait for the visitor to search. |

`distance.text` and `duration.text` are display strings such as "0.5 mi" and "3 mins". Use the numeric `value` fields for sorting. For their units, see [Distance and duration values](/api/routing#distance-and-duration-values).

### Errors and degraded paths

Every `Radar.*` method in this guide returns a promise that rejects with a `RadarError`. Read its `status`.

| Situation | What happens | What to do |
| --- | --- | --- |
| Location permission denied | `Radar.getLocation()` rejects with `ERROR_PERMISSIONS`. | Tell the visitor to search by address. Don't block the page. |
| Location unavailable or timed out | `Radar.getLocation()` rejects with `ERROR_LOCATION`. | Same as denied. |
| No locations near the origin | The search resolves with an empty array. | Show an empty state that names the searched place and invites another search. |
| Address has no suggestions | The autocomplete dropdown shows "No results". | Nothing. The visitor edits the text. |
| Wrong or revoked key | Requests reject with `ERROR_UNAUTHORIZED`, and the map fires its `error` event. | Show an unavailable state. Check the key and environment. |
| Rate limited | Requests reject with `ERROR_RATE_LIMIT`. | Ask the visitor to wait, and avoid issuing a search on every keystroke or map move. |
| Network failure or timeout | Requests reject with `ERROR_NETWORK` or another `RadarError`. | Show a retry message. A new search retries. |
| Route request fails | `Radar.distance()` or `Radar.matrix()` rejects. | Keep straight-line distances. Route distance is an enhancement. |
| A slow response arrives after a newer search | Two results race. | Discard the older response, as the example does with `searchCount`. |
| Map fails to load | The map's `error` event fires before `load`. | Keep the list working. The list is a complete fallback. See [Loading, errors, and recovery](/maps/maps#loading-errors-and-recovery). |
| Geofence missing from results | The location is disabled, has a different tag, is in the other environment, or is farther than `radius`. | Check the geofence in the dashboard for the environment that matches your key. |

On a phone, the example stacks the list above the map, each taking half the screen. If you hide the map behind a "Map" tab on small screens instead, call `map.resize()` when the tab is shown.

## Test the integration

1. **Manual search with no permission**: In your browser's site settings, block location for the page. Reload, type an address, and choose a suggestion. Results appear in the list and on the map. This is the path every visitor can use.
2. **List and map agree**: Count the list entries and the markers. They match, and the first entry is the nearest marker to the searched address.
3. **Select from the list**: Select an entry. It is highlighted, the map moves to it, its popup opens, and the distance line changes to a driving distance and time.
4. **Select from the map**: Select a different marker. The matching list entry is highlighted and scrolled into view, and only that marker's popup is open.
5. **Denied location**: With location still blocked, select "Use my location". The status line tells you to search by address, and address search still works.
6. **Granted location**: Allow location and select "Use my location". Results appear around your position.
7. **Empty results**: Search for a city where you have no locations. The empty state names the place you searched, and the next search works.
8. **No suggestions**: Type a string of random letters. The dropdown shows "No results" and nothing else changes.
9. **Search error**: In developer tools, block requests to `api.radar.io` and search again. The status line shows the error message. Unblock and search again to confirm recovery.
10. **Mobile layout**: At 360 pixels wide, the list and map are both visible and usable, and the search box is reachable without horizontal scrolling.
11. **Keys**: Search the page source and network requests for `_sk_`. There are no matches.

## Production checklist

- [ ] The page uses the Live publishable key, and your locations are imported into the Live environment with the same tags.
- [ ] No secret key appears in client code, the page source, or a client-side environment variable.
- [ ] Location data in Radar is kept current by sync or by API calls from your backend when a location opens, moves, or closes. Closed locations are disabled or deleted.
- [ ] Geofence metadata contains only information you intend to be public.
- [ ] Address search is the default path, and location permission is requested only from a user action.
- [ ] IP-based starting results are labeled as an estimate.
- [ ] Loading, empty, and error states exist for search, and the list works if the map fails.
- [ ] Straight-line and route distances are labeled, and the list and map use the same order.
- [ ] Searches are triggered by a selection or a button, not by every keystroke or map movement, to stay within rate limits.
- [ ] The layout is tested on a phone, with a keyboard, and with a screen reader. The status line uses `role="status"` so results are announced.
